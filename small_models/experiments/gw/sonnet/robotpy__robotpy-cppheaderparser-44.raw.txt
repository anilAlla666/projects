+import unittest
+
+import CppHeaderParser
+
+
+class TypedefReturnTypeTest(unittest.TestCase):
+    """Regression: method returning a typedef'd type raised KeyError: 'typedefs'"""
+
+    def test_method_returns_typedef_no_semicolon(self):
+        h = """
+typedef C A;
+
+class B
+{
+public:
+   A aMethod();
+}
+"""
+        cpp = CppHeaderParser.CppHeader(h, "string")
+        methods = cpp.classes["B"]["methods"]["public"]
+        self.assertEqual(len(methods), 1)
+        self.assertEqual(methods[0]["name"], "aMethod")
+        self.assertEqual(methods[0]["rtnType"], "A")
+
+    def test_method_returns_typedef_with_semicolon(self):
+        h = """
+typedef C A;
+
+class B
+{
+public:
+   A aMethod();
+};
+"""
+        cpp = CppHeaderParser.CppHeader(h, "string")
+        methods = cpp.classes["B"]["methods"]["public"]
+        self.assertEqual(methods[0]["name"], "aMethod")
