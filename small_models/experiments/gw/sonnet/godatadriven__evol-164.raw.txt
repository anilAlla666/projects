I added a new test file rather than editing an existing one, because I couldn't see the repository's current tests. I haven't run these tests and wrote them from memory of the `Population` API. They assume `Population(chromosomes=..., eval_function=...)`, `evaluate()`, `mutate(..., elitist=...)` and `individuals` with `.chromosome`.

`test_elitist_mutation_preserves_best` should fail on the current code, where the best individual is mutated anyway. After the fix it should pass. The other two tests cover non-elitist mutation and elitism with `maximize=False`.

+from evol import Population
+
+
+def _eval(x):
+    return x
+
+
+def test_elitist_mutation_preserves_best():
+    pop = Population(chromosomes=[1, 2, 3, 4], eval_function=_eval)
+    pop.evaluate()
+    pop.mutate(lambda x: x + 100, elitist=True)
+    chromosomes = sorted(ind.chromosome for ind in pop.individuals)
+    assert chromosomes == [4, 101, 102, 103]
+
+
+def test_non_elitist_mutation_mutates_all():
+    pop = Population(chromosomes=[1, 2, 3, 4], eval_function=_eval)
+    pop.evaluate()
+    pop.mutate(lambda x: x + 100)
+    chromosomes = sorted(ind.chromosome for ind in pop.individuals)
+    assert chromosomes == [101, 102, 103, 104]
+
+
+def test_elitist_mutation_minimize():
+    pop = Population(chromosomes=[1, 2, 3, 4], eval_function=_eval, maximize=False)
+    pop.evaluate()
+    pop.mutate(lambda x: x + 100, elitist=True)
+    chromosomes = sorted(ind.chromosome for ind in pop.individuals)
+    assert chromosomes == [1, 102, 103, 104]
