+import pytest

+import reframe.core.runtime as rt
+from reframe.core.backends import getscheduler
+from reframe.core.launchers.registry import getlauncher
+from reframe.core.schedulers import Job
+
+
+def _make_srun_job(tmp_path):
+    launcher = getlauncher('srun')()
+    return Job.create(getscheduler('local')(), launcher,
+                      name='fake_job',
+                      script_filename=str(tmp_path / 'job.sh'),
+                      stdout=str(tmp_path / 'job.out'),
+                      stderr=str(tmp_path / 'job.err'))
+
+
+def test_srun_passes_cpus_per_task(tmp_path):
+    # Since Slurm 22.05 srun does not read SLURM_CPUS_PER_TASK, so the
+    # launcher must pass --cpus-per-task explicitly
+    job = _make_srun_job(tmp_path)
+    job._num_tasks = 4
+    job._num_cpus_per_task = 4
+    cmd = job.launcher.command(job)
+    assert '--cpus-per-task=4' in cmd
+
+
+def test_srun_no_cpus_per_task_if_unset(tmp_path):
+    job = _make_srun_job(tmp_path)
+    job._num_tasks = 4
+    job._num_cpus_per_task = None
+    cmd = job.launcher.command(job)
+    assert not any(c.startswith('--cpus-per-task') for c in cmd)
