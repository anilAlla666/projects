The test writes two files with different type comments and runs `com2ann f1.py f2.py` on them. It then checks that each file is rewritten in place with its own annotations. With the bug, both inputs are written to the first file's path. `f1.py` would then hold `f2.py`'s content, and `f2.py` would stay unconverted. So the test should fail on the current code and pass once each input goes to its own output.

+import os
+import shutil
+import sys
+import tempfile
+import unittest
+from unittest import mock
+
+from com2ann import main
+
+
+class MultipleFilesTestCase(unittest.TestCase):
+    def setUp(self) -> None:
+        self.tmpdir = tempfile.mkdtemp()
+
+    def tearDown(self) -> None:
+        shutil.rmtree(self.tmpdir)
+
+    def _write(self, name: str, content: str) -> str:
+        path = os.path.join(self.tmpdir, name)
+        with open(path, "w") as f:
+            f.write(content)
+        return path
+
+    def _read(self, path: str) -> str:
+        with open(path) as f:
+            return f.read()
+
+    def test_multiple_input_files_written_separately(self) -> None:
+        f1 = self._write("f1.py", "x = 1  # type: int\n")
+        f2 = self._write("f2.py", "y = 'a'  # type: str\n")
+
+        with mock.patch.object(sys, "argv", ["com2ann", f1, f2]):
+            try:
+                main()
+            except SystemExit as e:
+                self.assertIn(e.code, (None, 0))
+
+        out1 = self._read(f1)
+        out2 = self._read(f2)
+        self.assertIn("x: int = 1", out1)
+        self.assertNotIn("y", out1)
+        self.assertIn("y: str = 'a'", out2)
+        self.assertNotIn("type:", out2)
+
+
+if __name__ == "__main__":
+    unittest.main()
