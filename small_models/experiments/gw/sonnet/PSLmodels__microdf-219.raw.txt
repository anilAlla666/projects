+import numpy as np
+import pytest
+
+import microdf as mdf
+
+
+def make_df():
+    return mdf.MicroDataFrame(
+        {"x": [1, 2, 3], "y": [1, 2, 2]}, weights=[4, 5, 6]
+    )
+
+
+def test_boolean_subset_weights_shape():
+    d = make_df()
+    d2 = d[d.y > 1]
+    assert d2.shape == (2, 2)
+    assert d2.weights.shape == (2,)
+
+
+def test_boolean_subset_weights_values():
+    d = make_df()
+    d2 = d[d.y > 1]
+    assert list(np.asarray(d2.weights)) == [5, 6]
+
+
+def test_subset_column_weights():
+    d = make_df()
+    d2 = d[d.y > 1]
+    assert list(np.asarray(d2.x.weights)) == [5, 6]
+
+
+def test_subset_mean():
+    d = make_df()
+    result = d[d.y > 1].mean()
+    assert float(result["x"]) == pytest.approx((2 * 5 + 3 * 6) / 11)
+    assert float(result["y"]) == pytest.approx(2.0)
+
+
+def test_subset_sum():
+    d = make_df()
+    result = d[d.y > 1].sum()
+    assert float(result["x"]) == pytest.approx(2 * 5 + 3 * 6)
+
+
+def test_column_subset_keeps_weights():
+    d = make_df()
+    d2 = d[["x"]]
+    assert d2.weights.shape == (3,)
+    assert list(np.asarray(d2.weights)) == [4, 5, 6]
+
+
+def test_microseries_unchanged():
+    s = mdf.MicroSeries([1, 2, 3], weights=[4, 5, 6])
+    s2 = s[s > 1]
+    assert s2.shape == (2,)
+    assert s2.weights.shape == (2,)
