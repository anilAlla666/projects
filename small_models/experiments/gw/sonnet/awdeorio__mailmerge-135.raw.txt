I couldn't see the repository, so I wrote this as a new test file, `tests/test_bom.py`, rather than appending to an existing one. I haven't run it. It assumes the CLI is a click command at `mailmerge.__main__.main` with `--template`, `--database`, `--config` and `--dry-run` options.

The test writes the issue's database with a UTF-8 BOM and CRLF line endings, then runs a dry-run merge. Without BOM handling, the first header comes back as `\ufeffname`, so `{{name}}` can't be resolved and the run fails. A correct fix should exit 0 and print "Hello My Name" with `TO: to@test.com`.

```diff
--- /dev/null
+++ b/tests/test_bom.py
@@ -0,0 +1,57 @@
+"""
+Tests for databases saved with a UTF-8 byte order mark (BOM).
+"""
+import click.testing
+from mailmerge.__main__ import main
+
+
+def _write_files(tmp_path, database_bytes):
+    template = tmp_path / "mailmerge_template.txt"
+    template.write_text(
+        "TO: {{email}}\n"
+        "FROM: My Self <myself@mydomain.com>\n"
+        "\n"
+        "Hello {{name}}\n"
+    )
+    database = tmp_path / "mailmerge_database.csv"
+    database.write_bytes(database_bytes)
+    config = tmp_path / "mailmerge_server.conf"
+    config.write_text(
+        "[smtp_server]\n"
+        "host = open-smtp.example.com\n"
+        "port = 25\n"
+    )
+    return template, database, config
+
+
+def _run(template, database, config):
+    runner = click.testing.CliRunner()
+    return runner.invoke(
+        main,
+        [
+            "--template", str(template),
+            "--database", str(database),
+            "--config", str(config),
+            "--dry-run",
+        ],
+    )
+
+
+def test_database_with_bom(tmp_path):
+    """First header key must be found when the CSV starts with a BOM."""
+    data = b"\xef\xbb\xbfname,email\r\nMy Name,to@test.com\r\n"
+    template, database, config = _write_files(tmp_path, data)
+    result = _run(template, database, config)
+    assert result.exception is None, result.output
+    assert result.exit_code == 0, result.output
+    assert "Hello My Name" in result.output
+    assert "to@test.com" in result.output
+
+
+def test_database_without_bom(tmp_path):
+    """Plain CSV keeps working."""
+    data = b"name,email\r\nMy Name,to@test.com\r\n"
+    template, database, config = _write_files(tmp_path, data)
+    result = _run(template, database, config)
+    assert result.exit_code == 0, result.output
+    assert "Hello My Name" in result.output
```
