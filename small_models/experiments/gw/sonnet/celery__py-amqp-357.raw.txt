+import ssl
+from unittest.mock import MagicMock, patch
+
+from amqp.transport import SSLTransport
+
+
+def _wrap(**kwargs):
+    """Call _wrap_socket_sni and return the SSLContext that was used."""
+    transport = object.__new__(SSLTransport)
+    sock = MagicMock()
+    with patch.object(ssl.SSLContext, 'wrap_socket',
+                      autospec=True) as wrap_socket:
+        transport._wrap_socket_sni(sock, **kwargs)
+    assert wrap_socket.called
+    return wrap_socket.call_args[0][0]
+
+
+class test_wrap_socket_sni_cert_reqs:
+
+    def test_cert_none_with_server_hostname(self):
+        # Previously raised:
+        # ValueError: Cannot set verify_mode to CERT_NONE when
+        # check_hostname is enabled.
+        context = _wrap(cert_reqs=ssl.CERT_NONE,
+                        server_hostname='example.com')
+        assert context.verify_mode == ssl.CERT_NONE
+        assert context.check_hostname is False
+
+    def test_cert_none_without_server_hostname(self):
+        context = _wrap(cert_reqs=ssl.CERT_NONE)
+        assert context.verify_mode == ssl.CERT_NONE
+        assert context.check_hostname is False
+
+    def test_cert_required_with_server_hostname(self):
+        context = _wrap(cert_reqs=ssl.CERT_REQUIRED,
+                        server_hostname='example.com')
+        assert context.verify_mode == ssl.CERT_REQUIRED
+        assert context.check_hostname is True
