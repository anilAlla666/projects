I added a new test file, `tests/test_validation_error_hash.py`. I have not run it. It should fail on the current code and pass once `ValidationError` is hashable.

The tests rely on `ValidationError` defining `__eq__` without `__hash__`. That makes it unhashable on every Python version, so `hash(error)` should raise `TypeError` today, not only on 3.6. The multi-message test also assumes `Message` is hashable. If the fix only hashes `ValidationError` and `Message` is not hashable, that test could still fail.

+import io
+import traceback

+import pytest

+import typesystem

+def _get_error(field, value):
+    with pytest.raises(typesystem.ValidationError) as exc_info:
+        field.validate(value)
+    return exc_info.value
+
+
+def test_validation_error_is_hashable():
+    error = _get_error(typesystem.Integer(), "hello")
+    assert isinstance(hash(error), int)
+    assert error in {error}
+
+
+def test_equal_validation_errors_have_equal_hashes():
+    error_a = _get_error(typesystem.Integer(), "hello")
+    error_b = _get_error(typesystem.Integer(), "hello")
+    assert error_a == error_b
+    assert hash(error_a) == hash(error_b)
+
+
+def test_multi_message_validation_error_is_hashable():
+    schema = typesystem.Schema(
+        fields={"a": typesystem.Integer(), "b": typesystem.Integer()}
+    )
+    error = _get_error(schema, {"a": "x", "b": "y"})
+    assert len(error.messages()) == 2
+    assert isinstance(hash(error), int)
+    error_again = _get_error(schema, {"a": "x", "b": "y"})
+    assert hash(error) == hash(error_again)
+
+
+def test_different_validation_errors_are_not_equal_in_set():
