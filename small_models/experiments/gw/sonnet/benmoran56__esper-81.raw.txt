I've added a new test file rather than editing an existing one, because I couldn't see the current contents of the repo's test files. I haven't run these tests against the current or fixed code.

The tests cover the behaviour the issue describes: an entity created with no components must exist, and the entity counter must not skip ids. They also check that a component can be added to such an entity later and that it can be deleted immediately.

I deliberately did not test that `add_component()` raises on an unknown entity. The issue only suggests that as something to consider, so a fix that keeps the current behaviour should still pass.

+import esper
+
+
+class Position:
+    def __init__(self, x=0, y=0):
+        self.x = x
+        self.y = y
+
+
+def test_create_entity_without_components_exists():
+    world = esper.World()
+    entity = world.create_entity()
+    assert world.entity_exists(entity)
+
+
+def test_create_entity_without_components_counter():
+    world = esper.World()
+    entity_1 = world.create_entity()
+    entity_2 = world.create_entity()
+    assert entity_1 == 1
+    assert entity_2 == 2
+    assert world.entity_exists(entity_1)
+    assert world.entity_exists(entity_2)
+
+
+def test_create_entity_with_components_exists():
+    world = esper.World()
+    entity = world.create_entity(Position(1, 2))
+    assert world.entity_exists(entity)
+    assert world.has_component(entity, Position)
+
+
+def test_empty_entity_can_receive_components_later():
+    world = esper.World()
+    entity = world.create_entity()
+    world.add_component(entity, Position(3, 4))
+    assert world.entity_exists(entity)
+    assert world.component_for_entity(entity, Position).x == 3
+
+
+def test_empty_entity_can_be_deleted():
+    world = esper.World()
+    entity = world.create_entity()
+    assert world.entity_exists(entity)
+    world.delete_entity(entity, immediate=True)
+    assert not world.entity_exists(entity)
+
+
+def test_nonexistent_entity_does_not_exist():
+    world = esper.World()
+    world.create_entity()
+    assert not world.entity_exists(999)
