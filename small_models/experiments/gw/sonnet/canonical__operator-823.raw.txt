I've added a new test file that builds `ops.model.Network` directly with a hostname in `ingress-addresses`, as the AWS ELB hostname in the issue produces. I haven't run it. On the current code `Network.__init__` raises `ValueError` for the hostname, so both tests should fail. They should pass once the hostname is kept as a plain string instead of raising.

I put the tests in a new file rather than `test/test_model.py` because I couldn't see that file's fixtures. The tests assume the fix keeps a non-IP ingress address as the raw `str` rather than dropping it.

+import ipaddress
+import unittest
+
+import ops.model
+
+HOSTNAME = 'acb396cf5563d429f9ebc8aa23ae47ed-1516332866.us-east-1.elb.amazonaws.com'
+
+
+class TestNetworkHostnameIngress(unittest.TestCase):
+    def _info(self, ingress):
+        return {
+            'bind-addresses': [{
+                'interface-name': 'eth0',
+                'addresses': [{'hostname': '', 'value': '10.1.0.1', 'cidr': '10.1.0.0/24'}],
+            }],
+            'egress-subnets': ['10.1.0.1/32'],
+            'ingress-addresses': ingress,
+        }
+
+    def test_hostname_ingress_address(self):
+        network = ops.model.Network(self._info([HOSTNAME]))
+        self.assertEqual(network.ingress_addresses, [HOSTNAME])
+        self.assertEqual(network.ingress_address, HOSTNAME)
+        self.assertEqual(str(network.bind_address), '10.1.0.1')
+
+    def test_mixed_ingress_addresses(self):
+        network = ops.model.Network(self._info(['10.1.0.1', HOSTNAME, 'fe80::1']))
+        self.assertEqual(len(network.ingress_addresses), 3)
+        self.assertEqual(network.ingress_addresses[0], ipaddress.ip_address('10.1.0.1'))
+        self.assertIsInstance(network.ingress_addresses[0], ipaddress.IPv4Address)
+        self.assertEqual(network.ingress_addresses[1], HOSTNAME)
+        self.assertIsInstance(network.ingress_addresses[1], str)
+        self.assertIsInstance(network.ingress_addresses[2], ipaddress.IPv6Address)
+
+
+if __name__ == '__main__':
+    unittest.main()
