+import os
+import unittest
+
+import yamlsettings
+from yamlsettings.yamldict import YAMLDict
+
+
+class UpdateFromEnvNonVariableCharsTest(unittest.TestCase):
+
+    def setUp(self):
+        self._added = []
+
+    def tearDown(self):
+        for key in self._added:
+            os.environ.pop(key, None)
+
+    def _setenv(self, key, value):
+        os.environ[key] = value
+        self._added.append(key)
+
+    def test_hyphen_key(self):
+        self._setenv('FOO_BAR', 'barbaz')
+        settings = YAMLDict({'foo-bar': None})
+        yamlsettings.update_from_env(settings)
+        self.assertEqual(settings['foo-bar'], 'barbaz')
+
+    def test_dot_key(self):
+        self._setenv('FOO_BAR_BAZ', 'qux')
+        settings = YAMLDict({'foo.bar': {'baz': None}})
+        yamlsettings.update_from_env(settings)
+        self.assertEqual(settings['foo.bar']['baz'], 'qux')
+
+    def test_nested_hyphen_keys(self):
+        self._setenv('OUTER_KEY_INNER_KEY', 'value')
+        settings = YAMLDict({'outer-key': {'inner-key': None}})
+        yamlsettings.update_from_env(settings)
+        self.assertEqual(settings['outer-key']['inner-key'], 'value')
+
+    def test_plain_key_still_works(self):
+        self._setenv('PLAIN_KEY', 'ok')
+        settings = YAMLDict({'plain_key': None})
+        yamlsettings.update_from_env(settings)
+        self.assertEqual(settings['plain_key'], 'ok')
+
+
+if __name__ == '__main__':
+    unittest.main()
