+import pandas as pd
+from pytest import approx
+
+from lenskit.metrics.topn import ndcg
+
+
+def _truth():
+    return pd.DataFrame({'item': [1, 2, 3], 'rating': [3.0, 5.0, 4.0]}).set_index('item')
+
+
+def test_ndcg_no_k_truncates_truth():
+    # recs shorter than truth: ideal DCG must use only len(recs) items
+    truth = _truth()
+    recs = pd.DataFrame({'item': [1, 2]})
+
+    assert ndcg(recs, truth, k=2) == approx(8.0 / 9.0)
+    assert ndcg(recs, truth) == approx(8.0 / 9.0)
+
+
+def test_ndcg_k_less_than_recs_perfect():
+    truth = _truth()
+    recs = pd.DataFrame({'item': [2, 3, 1]})
+
+    assert ndcg(recs, truth) == approx(1.0)
+    assert ndcg(recs[:2], truth, k=2) == approx(1.0)
+    assert ndcg(recs, truth, k=2) == approx(1.0)
+
+
+def test_ndcg_k_less_than_recs_imperfect():
+    truth = _truth()
+    recs = pd.DataFrame({'item': [1, 2, 3]})
+
+    assert ndcg(recs, truth, k=2) == approx(8.0 / 9.0)
+    assert ndcg(recs, truth, k=2) == approx(ndcg(recs[:2], truth, k=2))
+
+
+def test_ndcg_k_matches_truncated_recs():
+    truth = _truth()
+    recs = pd.DataFrame({'item': [3, 1, 2]})
+
+    for k in (1, 2, 3):
+        assert ndcg(recs, truth, k=k) == approx(ndcg(recs[:k], truth, k=k))
