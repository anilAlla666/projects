+from ioccontainer import provider, inject

+class NoneOverrideService(object):
+    pass
+
+
+@provider('none_override_service')
+def provide_none_override_service():
+    return NoneOverrideService()
+
+
+@inject(service='none_override_service')
+def kw_func(service=None):
+    return service
+
+
+@inject(service='none_override_service')
+def pos_func(service):
+    return service
+
+
+def test_none_kwarg_is_overridden():
+    result = kw_func(service=None)
+    assert isinstance(result, NoneOverrideService)
+
+
+def test_none_positional_is_overridden():
+    result = pos_func(None)
+    assert isinstance(result, NoneOverrideService)
+
+
+def test_not_provided_is_injected():
+    assert isinstance(kw_func(), NoneOverrideService)
+
+
+def test_non_none_value_is_kept():
+    sentinel = object()
+    assert kw_func(service=sentinel) is sentinel
+    assert pos_func(sentinel) is sentinel

+
