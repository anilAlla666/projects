+def _walk(obj):
+    """Yield all dicts found in a nested structure."""
+    if isinstance(obj, dict):
+        yield obj
+        for v in obj.values():
+            yield from _walk(v)
+    elif isinstance(obj, (list, tuple)):
+        for v in obj:
+            yield from _walk(v)
+
+
+class DataToolsUnitPlacementTest(TestCase):
+    def _run(self, method):
+        topic_d = {
+            "name": "Topic",
+            "fields": ["temperature"],
+            "units": ["°C"],
+            "data_tools": [
+                {"field": "temperature", "method": method},
+            ],

+        data = [

+        ]

+    def test_unit_is_inside_value_payload(self):
+        for method in ("average", "min", "max", "sum"):
