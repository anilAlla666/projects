I'm adding a new test file, `tests/unit/v1/test_transforms_copy.py`. I haven't run it. It checks that `Sentinel` keeps its identity through `copy.copy`, `copy.deepcopy` and `dataclasses.asdict`.

The issue's proposed `__deepcopy__(self)` takes no `memo` argument, so `copy.deepcopy` would call it with `memo` and raise a `TypeError`. A correct fix needs `__deepcopy__(self, memo)`, and the tests would catch the no-argument version.

I used `import dataclasses` and the sentinels `SERVER_TIMESTAMP`, `DELETE_FIELD`, `ArrayUnion`-style sentinels are not needed. The tests cover `SERVER_TIMESTAMP` and `DELETE_FIELD`, plus nested containers and a custom `Sentinel`.

+import copy
+import dataclasses
+
+from google.cloud.firestore_v1.transforms import (
+    DELETE_FIELD,
+    SERVER_TIMESTAMP,
+    Sentinel,
+)
+
+
+def test_sentinel_copy_preserves_identity():
+    assert copy.copy(SERVER_TIMESTAMP) is SERVER_TIMESTAMP
+    assert copy.copy(DELETE_FIELD) is DELETE_FIELD
+
+
+def test_sentinel_deepcopy_preserves_identity():
+    assert copy.deepcopy(SERVER_TIMESTAMP) is SERVER_TIMESTAMP
+    assert copy.deepcopy(DELETE_FIELD) is DELETE_FIELD
+
+
+def test_custom_sentinel_deepcopy_preserves_identity():
+    sentinel = Sentinel("custom")
+    assert copy.copy(sentinel) is sentinel
+    assert copy.deepcopy(sentinel) is sentinel
+
+
+def test_deepcopy_nested_container_preserves_identity():
+    data = {"a": SERVER_TIMESTAMP, "b": [DELETE_FIELD, {"c": SERVER_TIMESTAMP}]}
+    result = copy.deepcopy(data)
+    assert result["a"] is SERVER_TIMESTAMP
+    assert result["b"][0] is DELETE_FIELD
+    assert result["b"][1]["c"] is SERVER_TIMESTAMP
+
+
+def test_dataclass_asdict_preserves_sentinel():
+    @dataclasses.dataclass
+    class Example:
+        timestamp: object = SERVER_TIMESTAMP
+
+    example = Example()
+    assert example.timestamp is SERVER_TIMESTAMP
+    assert dataclasses.asdict(example)["timestamp"] is SERVER_TIMESTAMP
