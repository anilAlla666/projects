This diff adds one test file that checks the `Reach` constructor keeps the input dtype when it adds an `<UNK>` token. I haven't run it against the repository. It assumes the constructor takes an `unk_token` keyword and exposes `vectors` and `unk_index`, which I recalled from memory rather than checking.

The current code builds the `<UNK>` row with `np.zeros`, which defaults to float64. Concatenating that with float32 vectors upcasts the whole array to float64, so the float32 and float16 tests should fail today and pass once the fix preserves the dtype.

I tested the constructor only, not `Reach.load`. I wasn't sure of the exact `load` signature, and a wrong guess would make the tests fail for the wrong reason.

+import unittest
+
+import numpy as np
+
+from reach import Reach
+
+
+class TestUnkDtype(unittest.TestCase):
+    def _make(self, dtype, unk_token="<UNK>"):
+        vectors = np.random.RandomState(0).rand(3, 4).astype(dtype)
+        items = ["a", "b", "c"]
+        return vectors, Reach(vectors, items, unk_token=unk_token)
+
+    def test_float32_preserved_with_unk(self):
+        vectors, r = self._make(np.float32)
+        self.assertEqual(r.vectors.dtype, np.float32)
+        self.assertEqual(r.vectors.shape, (4, 4))
+
+    def test_float16_preserved_with_unk(self):
+        _, r = self._make(np.float16)
+        self.assertEqual(r.vectors.dtype, np.float16)
+
+    def test_float64_preserved_with_unk(self):
+        _, r = self._make(np.float64)
+        self.assertEqual(r.vectors.dtype, np.float64)
+
+    def test_no_unk_preserves_dtype(self):
+        vectors = np.ones((3, 4), dtype=np.float32)
+        r = Reach(vectors, ["a", "b", "c"])
+        self.assertEqual(r.vectors.dtype, np.float32)
+
+    def test_unk_vector_is_zero_and_original_kept(self):
+        vectors, r = self._make(np.float32)
+        self.assertTrue(np.all(r.vectors[r.unk_index] == 0))
+        self.assertTrue(np.allclose(r.vectors[1:], vectors))
+
+
+if __name__ == "__main__":
+    unittest.main()
