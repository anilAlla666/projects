+"""Tests for ZeroDivisionError when all data values are zero."""
+import contextlib
+import io
+import unittest
+
+from termgraph import termgraph
+
+
+def _flatten(data):
+    out = []
+    for row in data:
+        if isinstance(row, (list, tuple)):
+            out.extend(row)
+        else:
+            out.append(row)
+    return out
+
+
+class ZeroDataTests(unittest.TestCase):
+    def test_normalize_all_zeros(self):
+        try:
+            result = termgraph.normalize([[0], [0]], 100)
+        except ZeroDivisionError:
+            self.fail("normalize raised ZeroDivisionError on all-zero data")
+        self.assertTrue(all(v == 0 for v in _flatten(result)))
+
+    def test_normalize_all_zeros_multiple_series(self):
+        try:
+            result = termgraph.normalize([[0, 0], [0, 0]], 50)
+        except ZeroDivisionError:
+            self.fail("normalize raised ZeroDivisionError on all-zero data")
+        self.assertTrue(all(v == 0 for v in _flatten(result)))
+
+    def test_normalize_all_zero_floats(self):
+        try:
+            termgraph.normalize([[0.0], [0.0]], 100)
+        except ZeroDivisionError:
+            self.fail("normalize raised ZeroDivisionError on all-zero data")
+
+    def test_chart_all_zeros(self):
+        args = {
+            'width': 100, 'format': '{:,}', 'suffix': '', 'no_labels': False,
+            'no_values': False, 'vertical': False, 'stacked': False,
+            'histogram': False, 'different_scale': False,
+            'title': None, 'color': None, 'label_before': False,
+            'delim': '', 'verbose': False, 'custom_tick': '', 'tick': None,
+            'space_between': False, 'calendar': False, 'start_dt': None,
+            'end_color': None, 'percentage': False, 'no_readable': False,

+        }

+        try:
+            with contextlib.redirect_stdout(buf):
+                termgraph.chart(
+                    colors=[],
+                    data=[[0], [0]],
+                    args=args,
+                    labels=['iOS', 'Android'],
+                )
+        except ZeroDivisionError:
