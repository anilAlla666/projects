This test file checks that a line break in `additional_information` no longer adds extra lines to the QR payload. I haven't run it against the repository. It assumes `QRBill.qr_data()` returns the payload as a string and that the constructor accepts the usual `account`, `creditor` and `additional_information` arguments.

The tests compare the payload from a bill containing `"Hello \n Line break"` with the payload from a bill whose text has no line break. The payload must have the same number of lines, and both words must still appear. Carriage returns, `\r\n` and multiple breaks are checked the same way. The expected sanitized text is not asserted exactly, since the fix might replace a break with a space or collapse whitespace.

+import unittest
+
+from qrbill import QRBill
+
+
+class AdditionalInformationLineBreakTests(unittest.TestCase):
+    def _bill(self, info):
+        return QRBill(
+            account="CH 53 8000 5000 0102 83664",
+            creditor={
+                'name': 'Jane', 'pcode': '1000', 'city': 'Lausanne',
+            },
+            additional_information=info,
+        )
+
+    def _lines(self, info):
+        return self._bill(info).qr_data().splitlines()
+
+    def test_newline_does_not_add_lines_to_payload(self):
+        reference = self._lines("Hello  Line break")
+        lines = self._lines("Hello \n Line break")
+        self.assertEqual(len(lines), len(reference))
+
+    def test_newline_content_preserved(self):
+        data = self._bill("Hello \n Line break").qr_data()
+        self.assertIn("Hello", data)
+        self.assertIn("Line break", data)
+
+    def test_carriage_return_removed(self):
+        reference = self._lines("Hello  Line break")
+        for info in ("Hello \r\n Line break", "Hello \r Line break"):
+            lines = self._lines(info)
+            self.assertEqual(len(lines), len(reference), repr(info))
+
+    def test_multiple_newlines(self):
+        reference = self._lines("Hello World Again")
+        lines = self._lines("Hello\nWorld\n\nAgain")
+        self.assertEqual(len(lines), len(reference))
+        data = "\n".join(lines)
+        for word in ("Hello", "World", "Again"):
+            self.assertIn(word, data)
+
+    def test_words_not_glued_together(self):
+        data = self._bill("Hello\nWorld").qr_data()
+        self.assertNotIn("HelloWorld", data)
+
+    def test_no_newline_in_attribute(self):
+        bill = self._bill("Hello \n Line break")
+        info = bill.additional_information
+        self.assertNotIn("\n", info)
+        self.assertNotIn("\r", info)
+
+
+if __name__ == '__main__':
+    unittest.main()
