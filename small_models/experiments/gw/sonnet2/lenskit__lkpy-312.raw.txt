This adds one new test file covering both faults from the issue. I haven't run it against the current code. The expected values are computed by hand with the default `log2` discount, where positions 1 and 2 both get weight 1.

- **No `k`, recs shorter than truth:** for recs `[1, 2]` against ratings 3, 5, 4, the ideal DCG is truncated to the recommendation length, giving 5 + 4 = 9. The recommendation DCG is 3 + 5 = 8, so nDCG is 8/9 ≈ 0.8888.
- **`k` smaller than the recs list:** recs `[2, 3, 1]` with `k=2` should give a perfect 1.0, and `k=1` on recs `[1, 2, 3]` should give 3/5 = 0.6.

+import numpy as np
+import pandas as pd
+
+from pytest import approx
+
+from lenskit.metrics.topn import ndcg
+
+
+def _truth():
+    truth = pd.DataFrame({'item': [1, 2, 3], 'rating': [3.0, 5.0, 4.0]})
+    return truth.set_index('item')
+
+
+def test_ndcg_no_k_short_recs():
+    # recs shorter than truth, no k: ideal DCG must be truncated to rec length
+    truth = _truth()
+    recs = pd.DataFrame({'item': [1, 2]})
+
+    # ideal = 5 + 4 = 9, actual = 3 + 5 = 8
+    assert ndcg(recs, truth) == approx(8.0 / 9.0)
+    assert ndcg(recs, truth) == approx(ndcg(recs, truth, k=2))
+
+
+def test_ndcg_k_less_than_recs_perfect():
+    truth = _truth()
+    recs = pd.DataFrame({'item': [2, 3, 1]})
+
+    assert ndcg(recs, truth) == approx(1.0)
+    assert ndcg(recs, truth, k=2) == approx(1.0)
+    assert ndcg(recs[:2], truth, k=2) == approx(1.0)
+
+
+def test_ndcg_k_less_than_recs_matches_truncated_recs():
+    truth = _truth()
+    recs = pd.DataFrame({'item': [1, 3, 2]})
+
+    for k in (1, 2):
+        assert ndcg(recs, truth, k=k) == approx(ndcg(recs[:k], truth, k=k))
+
+
+def test_ndcg_k_one():
+    truth = _truth()
+    recs = pd.DataFrame({'item': [1, 2, 3]})
+
+    # actual = 3, ideal = 5
+    assert ndcg(recs, truth, k=1) == approx(0.6)
+
+
+def test_ndcg_k_less_than_recs_with_discount():
+    truth = _truth()
+    recs = pd.DataFrame({'item': [1, 2, 3]})
+
+    # k=3 positions: discounts 1, 1, log2(3)
+    # k=2 should only use first two positions: actual = 3 + 5 = 8, ideal = 9
+    assert ndcg(recs, truth, k=2) == approx(8.0 / 9.0)
+    full = (3 + 5 + 4 / np.log2(3)) / (5 + 4 + 3 / np.log2(3))
+    assert ndcg(recs, truth, k=3) == approx(full)
+    assert ndcg(recs, truth) == approx(full)
