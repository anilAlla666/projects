I haven't run this test or read `filesystem.py`, so it relies on my recollection of the `FakeFilesystem` API (`add()` plus the `root` attribute). It has two tests. One opens a `DirEntry` directly while an overlay is active. The other runs `shutil.copytree` from a real directory into an overlayed path. On the current code both should fail with `AttributeError: 'posix.DirEntry' object has no attribute 'startswith'`, and they should pass once `_is_fake_path` handles `DirEntry` arguments. The `copytree` test only exercises the bug on Python 3.8+, where `shutil` passes `DirEntry` objects to `open`.

+import os
+import shutil
+import tempfile
+
+from testtools import TestCase
+
+from systemfixtures import FakeFilesystem
+
+
+class DirEntryTest(TestCase):
+
+    def setUp(self):
+        super(DirEntryTest, self).setUp()
+        self.filesystem = self.useFixture(FakeFilesystem())
+        self.src = tempfile.mkdtemp()
+        self.addCleanup(shutil.rmtree, self.src, True)
+
+    def test_open_direntry(self):
+        """open() accepts a DirEntry while an overlay is active."""
+        self.filesystem.add("/foo")
+        with open(os.path.join(self.src, "a.txt"), "w") as fd:
+            fd.write("hello")
+        entries = list(os.scandir(self.src))
+        self.assertEqual(1, len(entries))
+        with open(entries[0], "rb") as fd:
+            self.assertEqual(b"hello", fd.read())
+
+    def test_copytree_to_overlayed_dir(self):
+        """shutil.copytree works with an overlayed destination."""
+        self.filesystem.add("/foo")
+        with open(os.path.join(self.src, "a.txt"), "w") as fd:
+            fd.write("hello")
+        os.mkdir(os.path.join(self.src, "sub"))
+        with open(os.path.join(self.src, "sub", "b.txt"), "w") as fd:
+            fd.write("world")
+
+        shutil.copytree(self.src, "/foo/dst")
+
+        with open("/foo/dst/a.txt") as fd:
+            self.assertEqual("hello", fd.read())
+        with open("/foo/dst/sub/b.txt") as fd:
+            self.assertEqual("world", fd.read())
+        # The data really went to the overlay, not the real root.
+        self.assertTrue(
+            os.path.exists(
+                os.path.join(self.filesystem.root, "foo", "dst", "a.txt")))
