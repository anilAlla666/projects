This test calls `SSLTransport._wrap_socket_sni` with `cert_reqs=ssl.CERT_NONE` and checks that no `ValueError` is raised and the context ends up with `CERT_NONE`. On the current code, `verify_mode` is set while `check_hostname` is still `True`, so the first assertion path raises. I haven't run it against the repository. It patches `ssl.SSLContext.wrap_socket` and captures the context, so no real connection or handshake happens.

+import ssl
+from unittest.mock import MagicMock, patch
+
+from amqp.transport import SSLTransport
+
+
+def _wrap(cert_reqs, server_hostname):
+    transport = SSLTransport('localhost', 10)
+    captured = {}
+
+    def fake_wrap_socket(self, *args, **kwargs):
+        captured['context'] = self
+        return MagicMock()
+
+    with patch.object(ssl.SSLContext, 'wrap_socket', fake_wrap_socket):
+        transport._wrap_socket_sni(
+            MagicMock(),
+            cert_reqs=cert_reqs,
+            server_hostname=server_hostname,
+        )
+    return captured['context']
+
+
+class test_SSLTransport_cert_reqs_none:
+
+    def test_cert_none_with_server_hostname(self):
+        context = _wrap(ssl.CERT_NONE, 'localhost')
+        assert context.verify_mode == ssl.CERT_NONE
+        assert context.check_hostname is False
+
+    def test_cert_none_without_server_hostname(self):
+        context = _wrap(ssl.CERT_NONE, None)
+        assert context.verify_mode == ssl.CERT_NONE
+        assert context.check_hostname is False
+
+    def test_cert_required_with_server_hostname(self):
+        context = _wrap(ssl.CERT_REQUIRED, 'localhost')
+        assert context.verify_mode == ssl.CERT_REQUIRED
+        assert context.check_hostname is True
