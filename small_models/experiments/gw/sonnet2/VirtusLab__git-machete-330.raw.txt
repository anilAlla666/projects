+import tempfile
+import unittest
+
+REPO_ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))

+
+class AddUnmanagedBranchTest(unittest.TestCase):
+    def setUp(self):
+        self.dir = tempfile.mkdtemp()
+        self.env = dict(os.environ)
+        self.env["PYTHONPATH"] = REPO_ROOT + os.pathsep + self.env.get("PYTHONPATH", "")
+        self.env.update({
+            "GIT_AUTHOR_NAME": "Test", "GIT_AUTHOR_EMAIL": "t@example.com",
+            "GIT_COMMITTER_NAME": "Test", "GIT_COMMITTER_EMAIL": "t@example.com",
+        })
+        self.git("init")
+        self.git("checkout", "-b", "master")
+        self.commit("master commit")
+        self.git("checkout", "-b", "foo/bar")
+        self.commit("bar commit")
+        self.git("checkout", "master")
+        with open(os.path.join(self.dir, ".git", "machete"), "w") as f:
+            f.write("master\n")

+    def tearDown(self):

+    def git(self, *args):
+        subprocess.check_output(("git",) + args, cwd=self.dir, env=self.env,
+                                stderr=subprocess.STDOUT)
+

+        self.git("commit", "--allow-empty", "-m", msg)
+
+    def machete(self, *args):
+        code = ("import sys; from git_machete import cli; "
+                "sys.argv = ['git-machete'] + sys.argv[1:]; cli.main()")
+        p = subprocess.Popen([sys.executable, "-c", code] + list(args),
+                             cwd=self.dir, env=self.env,
+                             stdout=subprocess.PIPE, stderr=subprocess.PIPE,
+                             universal_newlines=True)
+        out, err = p.communicate()
+        return p.returncode, out, err
+
+    def test_add_existing_unmanaged_branch_with_slash(self):
+        rc, out, err = self.machete("add", "-y", "foo/bar")
+        self.assertNotIn("AttributeError", err)

+        self.assertEqual(rc, 0, msg=out + err)
