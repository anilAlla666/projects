I couldn't see `charmcraft/main.py`, so this test is written against assumptions about how it logs, and I haven't run it. It runs `main(["charmcraft", "version", "--verbose"])` with `CHARMCRAFT_AUTH` set to a secret. It collects output from `caplog`, stdout and stderr, then checks three things:

- `CHARMCRAFT_AUTH='<hidden>'` appears in the output.
- The secret value never appears.
- Other `CHARMCRAFT_*` variables are still logged with their values.

Two assumptions could make it fail or pass for the wrong reason: that the "System details" line reaches one of those three streams (it won't if it only goes to a log file), and that the `version` command with `--verbose` is accepted by the CLI.

+"""Tests that the presence of CHARMCRAFT_AUTH is logged without leaking it."""
+
+import logging
+
+from charmcraft import main as charmcraft_main
+
+SECRET = "super-secret-credentials-value"
+
+
+def _run_main(capsys, caplog):
+    caplog.set_level(logging.DEBUG)
+    try:
+        charmcraft_main.main(["charmcraft", "version", "--verbose"])
+    except SystemExit:
+        pass
+    captured = capsys.readouterr()
+    return caplog.text + captured.out + captured.err
+
+
+def test_auth_presence_is_logged_hidden(monkeypatch, capsys, caplog):
+    monkeypatch.setenv("CHARMCRAFT_AUTH", SECRET)
+    output = _run_main(capsys, caplog)
+
+    assert "CHARMCRAFT_AUTH='<hidden>'" in output
+    assert SECRET not in output
+
+
+def test_other_charmcraft_vars_still_logged(monkeypatch, capsys, caplog):
+    monkeypatch.setenv("CHARMCRAFT_AUTH", SECRET)
+    monkeypatch.setenv("CHARMCRAFT_SOMETHING_ELSE", "visible-value")
+    output = _run_main(capsys, caplog)
+
+    assert "CHARMCRAFT_SOMETHING_ELSE='visible-value'" in output
+    assert "CHARMCRAFT_AUTH='<hidden>'" in output
+    assert SECRET not in output
+
+
+def test_auth_not_logged_if_absent(monkeypatch, capsys, caplog):
+    monkeypatch.delenv("CHARMCRAFT_AUTH", raising=False)
+    output = _run_main(capsys, caplog)
+
+    assert "CHARMCRAFT_AUTH" not in output
