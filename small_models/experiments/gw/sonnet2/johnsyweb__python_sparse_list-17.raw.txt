I haven't run this against the repo. The diff adds `test_delete_empty_slice.py`, which checks that deleting an empty slice from a `SparseList` leaves it unchanged. It covers the issue's `del sl[3:3]` case plus empty slices at the start, at the end, past the end, and with reversed bounds. It also checks that a non-empty slice delete still works.

I'm assuming the failing path is the same for all of these (an `IndexError` when the slice is empty), so each empty-slice test should fail on the current code. I used `range` rather than `xrange` so the tests run on Python 2 and 3.

```diff

+import unittest
+
+import sparse_list
+
+
+class TestDeleteEmptySlice(unittest.TestCase):
+    def setUp(self):
+        self.sl = sparse_list.SparseList(list(range(5)), None)
+
+    def assert_unchanged(self):
+        self.assertEqual(5, len(self.sl))
+        for i in range(5):
+            self.assertEqual(i if i else 0, self.sl[i])
+
+    def test_delete_empty_slice_in_middle(self):
+        del self.sl[3:3]
+        self.assert_unchanged()
+
+    def test_delete_empty_slice_at_start(self):
+        del self.sl[0:0]
+        self.assert_unchanged()
+
+    def test_delete_empty_slice_at_end(self):
+        del self.sl[5:5]
+        self.assert_unchanged()
+
+    def test_delete_empty_slice_past_end(self):
+        del self.sl[10:10]
+        self.assert_unchanged()
+
+    def test_delete_empty_slice_reversed_bounds(self):
+        del self.sl[4:2]
+        self.assert_unchanged()
+
+    def test_delete_empty_slice_matches_list_behaviour(self):
+        expected = list(range(5))
+        del expected[3:3]
+        del self.sl[3:3]
+        self.assertEqual(len(expected), len(self.sl))
+        for i, v in enumerate(expected):
+            self.assertEqual(v, self.sl[i])
+
+    def test_delete_empty_slice_on_empty_list(self):
+        sl = sparse_list.SparseList(0, None)
+        del sl[0:0]
+        self.assertEqual(0, len(sl))
+
+    def test_delete_nonempty_slice_still_works(self):
+        del self.sl[1:3]
+        self.assertEqual(3, len(self.sl))
+        self.assertEqual(0, self.sl[0])
+        self.assertEqual(3, self.sl[1])
+        self.assertEqual(4, self.sl[2])
+
+
+if __name__ == '__main__':
+    unittest.main()
```
