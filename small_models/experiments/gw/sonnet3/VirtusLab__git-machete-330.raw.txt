I couldn't see the repo's existing test layout, so this adds a standalone test file that runs the CLI in a subprocess against a temporary git repo. I haven't run it. The first test reproduces the issue's crash (`git machete add foo/bar` on an existing, unmanaged branch), so it should fail on the current code. The second test checks that the same flow works for a branch name without a slash.

I'm assuming `git_machete.cli.main()` reads `sys.argv[1:]`, which is how I remember it. The tests also assume the project root is the parent of `tests/`, so that `import git_machete` resolves via `PYTHONPATH`.

+import os
+import shutil
+import subprocess
+import sys
+import tempfile
+import unittest
+
+ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
+
+
+class TestAddBranchArg(unittest.TestCase):
+    def setUp(self) -> None:
+        self.dir = tempfile.mkdtemp()
+        self.env = dict(os.environ)
+        self.env.update({
+            "GIT_AUTHOR_NAME": "Test", "GIT_AUTHOR_EMAIL": "t@example.com",
+            "GIT_COMMITTER_NAME": "Test", "GIT_COMMITTER_EMAIL": "t@example.com",
+            "PYTHONPATH": ROOT + os.pathsep + os.environ.get("PYTHONPATH", ""),
+        })
+        self.git("init", "-q")
+        self.git("checkout", "-q", "-b", "master")
+        self.commit("a")
+        self.git("checkout", "-q", "-b", "develop")
+        self.commit("b")
+        self.git("checkout", "-q", "-b", "foo/bar")
+        self.commit("c")
+        self.git("checkout", "-q", "master")
+        with open(os.path.join(self.dir, ".git", "machete"), "w") as f:
+            f.write("master\n")
+
+    def tearDown(self) -> None:
+        shutil.rmtree(self.dir, ignore_errors=True)
+
+    def git(self, *args: str) -> None:
+        subprocess.check_call(["git", *args], cwd=self.dir, env=self.env)
+
+    def commit(self, name: str) -> None:
+        with open(os.path.join(self.dir, name), "w") as f:
+            f.write(name)
+        self.git("add", name)
+        self.git("commit", "-q", "-m", name)
+
+    def machete(self, *args: str) -> "subprocess.CompletedProcess[str]":
+        code = "from git_machete import cli; cli.main()"
+        return subprocess.run(
+            [sys.executable, "-c", code, *args],
+            cwd=self.dir, env=self.env, input="y\n" * 5,
+            stdout=subprocess.PIPE, stderr=subprocess.PIPE,
+            universal_newlines=True)
+
+    def test_add_existing_unmanaged_branch_by_name(self) -> None:
+        result = self.machete("add", "foo/bar")
+        self.assertNotIn("AttributeError", result.stderr)
+        self.assertNotIn("Traceback", result.stderr)
+        self.assertEqual(result.returncode, 0, result.stderr)
+        with open(os.path.join(self.dir, ".git", "machete")) as f:
+            self.assertIn("foo/bar", f.read())
+
+    def test_add_existing_unmanaged_branch_without_slash(self) -> None:
+        result = self.machete("add", "develop")
+        self.assertNotIn("Traceback", result.stderr)
+        self.assertEqual(result.returncode, 0, result.stderr)
+        with open(os.path.join(self.dir, ".git", "machete")) as f:
+            self.assertIn("develop", f.read())
+
+
+if __name__ == "__main__":
+    unittest.main()
