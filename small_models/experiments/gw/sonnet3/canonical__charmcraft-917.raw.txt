+"""Tests for logging the presence of the auth environment variable."""

+from charmcraft.commands.store.client import ALTERNATE_AUTH_ENV_VAR

+
+def test_system_details_auth_present_is_hidden(monkeypatch):
+    """The presence of the auth variable is logged, but not its value."""
+    secret = "super-secret-credentials-123"
+    monkeypatch.setenv(ALTERNATE_AUTH_ENV_VAR, secret)
+
+    result = _get_system_details()
+
+    assert secret not in result
+    assert f"{ALTERNATE_AUTH_ENV_VAR}='<hidden>'" in result

+
+def test_system_details_auth_absent(monkeypatch):
+    """If the auth variable is not set, it is not mentioned."""
+    monkeypatch.delenv(ALTERNATE_AUTH_ENV_VAR, raising=False)
+
+    result = _get_system_details()
+
+    assert ALTERNATE_AUTH_ENV_VAR not in result
+
+
+def test_system_details_other_charmcraft_vars_still_logged(monkeypatch):
+    """Other CHARMCRAFT variables are logged with their real values."""
+    monkeypatch.setenv("CHARMCRAFT_FOO", "bar")
+    monkeypatch.setenv(ALTERNATE_AUTH_ENV_VAR, "secret")
+
+    result = _get_system_details()
+
+    assert "CHARMCRAFT_FOO='bar'" in result
