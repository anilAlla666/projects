import numpy as np
import porespy as ps
import openpnm as op
ws = op.Workspace()
ws.settings['loglevel'] = 50
ps.settings.tqdm['disable'] = True


class Snow2BoundaryWidthTest:

    def setup_class(self):
        np.random.seed(0)
        self.im = ps.generators.blobs(shape=[100, 100], porosity=0.6,
                                      blobiness=1.5)

    def test_asymmetric_pairs_on_all_axes_shape(self):
        im = self.im
        out = ps.networks.snow2(im, boundary_width=[[5, 15], [10, 10]],
                                parallelization=None)
        assert out.regions.shape == (im.shape[0] + 20, im.shape[1] + 20)

    def test_pairs_same_as_mixed_spec(self):
        im = self.im
        a = ps.networks.snow2(im, boundary_width=[[5, 15], 10],
                              parallelization=None)
        b = ps.networks.snow2(im, boundary_width=[[5, 15], [10, 10]],
                              parallelization=None)
        assert a.regions.shape == b.regions.shape
        assert np.all(a.regions == b.regions)
