+def test_symbol_model_rejects_reader_macro_syntax():
+    # `#foo` parses as a reader-macro call, not a symbol, so it can't
+    # be made into a `Symbol`, just like "5".
+    with pytest.raises(ValueError):
+        Symbol("5")
+    with pytest.raises(ValueError):
+        Symbol("#foo")
+    with pytest.raises(ValueError):
+        Symbol("#a")
+
+    # A `#` that isn't at the start is fine.
+    assert Symbol("foo#bar") == Symbol("foo#bar")
