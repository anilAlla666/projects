I haven't run these tests, and the root cause is my guess. The complement table that `reverseComplement` uses is built from `ambiguous_dna_complement`, which as I recall has only uppercase keys. If so, lowercase bases pass through unchanged on the reverse strand and are mistranslated, even though forward-strand translation works. Because the issue doesn't say whether a fix should preserve case in the complement, the tests compare results case-insensitively against the uppercase sequence.

from unittest import TestCase

from dark.reads import DNARead, RNARead


class TestLowercaseTranslation(TestCase):
    """
    Lowercase (and mixed case) nucleotides must be handled the same as
    uppercase ones.
    """
    SEQ = 'ACGTTGCAAGGCTTACGATCGGATCCATGAAATTTGGGCCC'

    def _translations(self, seq):
        return [(t.frame, t.reverseComplemented, t.sequence.upper())
                for t in DNARead('id', seq).translations()]

    def testLowercaseTranslationsMatchUppercase(self):
        self.assertEqual(self._translations(self.SEQ),
                         self._translations(self.SEQ.lower()))

    def testMixedCaseTranslationsMatchUppercase(self):
        mixed = ''.join(c.lower() if i % 2 else c
                        for i, c in enumerate(self.SEQ))
        self.assertEqual(self._translations(self.SEQ),
                         self._translations(mixed))

    def testPartiallyLowercaseTranslationsMatchUppercase(self):
        mixed = self.SEQ[:10].lower() + self.SEQ[10:]
        self.assertEqual(self._translations(self.SEQ),
                         self._translations(mixed))

    def testReverseComplementLowercase(self):
        self.assertEqual(
            DNARead('id', 'ACGTNRY').reverseComplement().sequence.upper(),
            DNARead('id', 'acgtnry').reverseComplement().sequence.upper())

    def testReverseComplementMixedCase(self):
        self.assertEqual(
            DNARead('id', 'AACCGGTT').reverseComplement().sequence.upper(),
            DNARead('id', 'aaCCggTT').reverseComplement().sequence.upper())

    def testReverseComplementValueLowercase(self):
        self.assertEqual(
            'AACCGGTT',
            DNARead('id', 'aaccggtt').reverseComplement().sequence.upper())

    def testRNAReverseComplementLowercase(self):
        self.assertEqual(
            RNARead('id', 'ACGU').reverseComplement().sequence.upper(),
            RNARead('id', 'acgu').reverseComplement().sequence.upper())
