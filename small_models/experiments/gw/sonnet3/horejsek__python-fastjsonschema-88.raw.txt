+import pytest

+from fastjsonschema import JsonSchemaException, compile

+
+def test_type_array_accepts_tuple():
+    validate = compile({'type': 'array'})
+    assert validate([1, 2]) == [1, 2]
+    assert validate((1, 2)) == (1, 2)
+    assert validate(()) == ()
+
+
+def test_type_array_among_multiple_types():
+    validate = compile({'type': ['array', 'null']})
+    validate((1, 2))
+    validate(None)
+    with pytest.raises(JsonSchemaException):
+        validate('abc')
+
+
+@pytest.mark.parametrize('schema_type', ['string', 'object', 'number', 'integer', 'boolean', 'null'])
+def test_tuple_is_not_other_types(schema_type):
+    validate = compile({'type': schema_type})
+    with pytest.raises(JsonSchemaException):
+        validate((1, 2))
+
+
+def test_items_single_schema():
+    validate = compile({'type': 'array', 'items': {'type': 'integer'}})
+    validate((1, 2, 3))
+    with pytest.raises(JsonSchemaException):

+def test_items_without_type():
