+def test_scan_iter_type_filters_mixed_keys(r):
+    r.set("str1", "v")
+    r.zadd("zset1", {"a": 1})
+    r.sadd("set1", "a")
+    r.rpush("list1", "a")
+    r.hset("hash1", "f", "v")
+    assert set(r.scan_iter(_type="STRING")) == {b"str1"}
+    assert set(r.scan_iter(_type="ZSET")) == {b"zset1"}
+    assert set(r.scan_iter(_type="SET")) == {b"set1"}
+    assert set(r.scan_iter(_type="LIST")) == {b"list1"}
+    assert set(r.scan_iter(_type="HASH")) == {b"hash1"}
+
+
+def test_scan_type_option(r):
+    r.set("str1", "v")
+    r.zadd("zset1", {"a": 1})
