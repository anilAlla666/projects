I've added a new test file that checks the `get-details` request includes `type=api` alongside the line `id`. I haven't run it. It should fail on the current code, which sends only `id`, and pass once `type=api` is added.

The test does not depend on parameter order or the exact URL string. It parses the query string of the recorded request and checks the values. It relies on the existing `successful_login` and `mock_line_details` fixtures that `tests/test_api.py` already uses.

+from urllib.parse import urlparse, parse_qs
+
+import responses
+
+from redpocket import RedPocket
+
+
+def _details_calls():
+    return [
+        c
+        for c in responses.calls
+        if urlparse(c.request.url).path == "/account/get-details"
+    ]
+
+
+@responses.activate
+def test_get_line_details_requests_type_api(
+    successful_login: None, mock_line_details: dict
+):
+    responses.add(
+        responses.GET,
+        "https://www.redpocket.com/account/get-details",
+        status=200,
+        json={"return_code": 1, "return_data": mock_line_details},
+    )
+    rp = RedPocket(username="fake", password="password")
+    details = rp.get_line_details("AAA1BBB2")
+
+    assert details.number == 1234567890
+
+    calls = _details_calls()
+    assert len(calls) == 1
+    query = parse_qs(urlparse(calls[0].request.url).query)
+    assert query.get("id") == ["AAA1BBB2"]
+    assert query.get("type") == ["api"]
