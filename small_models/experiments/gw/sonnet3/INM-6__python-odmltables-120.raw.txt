The tests below cover a csv file that contains only a header. Loading it currently hits the undefined `row_id`. I haven't run them, and they rely on three assumptions about the `odmltables` API:

- `OdmlCsvTable.write2file` writes the column header as the first line of the csv.
- `OdmlTable.load_from_csv_table(path)` is the loading method.
- The loaded rows end up in `_odmldict`.

```diff
--- /dev/null

+# -*- coding: utf-8 -*-
+"""
+Tests for loading a csv file that only contains a header.
+"""
+import os
+import unittest
+
+from odmltables.odml_table import OdmlTable
+from odmltables.odml_csv_table import OdmlCsvTable
+
+from .create_test_odmls import create_small_test_odml
+
+
+class TestLoadEmptyCsv(unittest.TestCase):
+    def setUp(self):
+        self.filename = 'testtable_empty_header_only.csv'
+        table = OdmlCsvTable()
+        table.load_from_function(create_small_test_odml)
+        table.write2file(self.filename)
+
+        with open(self.filename) as f:
+            header = f.readline()
+        with open(self.filename, 'w') as f:
+            f.write(header)
+
+    def tearDown(self):
+        if os.path.exists(self.filename):
+            os.remove(self.filename)
+
+    def test_load_header_only_csv(self):
+        table = OdmlTable()
+        # must not raise (formerly: undefined 'row_id')
+        table.load_from_csv_table(self.filename)
+        self.assertEqual(len(table._odmldict), 0)
+
+    def test_load_header_only_csv_csvtable(self):
+        table = OdmlCsvTable()
+        table.load_from_csv_table(self.filename)
+        self.assertEqual(len(table._odmldict), 0)
+
+    def test_load_header_only_csv_with_trailing_newlines(self):
+        with open(self.filename, 'a') as f:
+            f.write('\n')
+        table = OdmlTable()
+        table.load_from_csv_table(self.filename)
+        self.assertEqual(len(table._odmldict), 0)
+
+
+if __name__ == '__main__':
+    unittest.main()
```
