+# coding: utf-8
+"""Exporting a parsed 'exists' test through the factory."""
+import unittest
+
+from sievelib.factory import FiltersSet
+from sievelib.parser import Parser
+
+
+class ExistsExportTestCase(unittest.TestCase):
+    def _load(self, script):
+        p = Parser()
+        self.assertTrue(p.parse(script), p.error)
+        fs = FiltersSet("test")
+        fs.from_parser_result(p)
+        return fs
+
+    def test_exists_single_header_list(self):
+        script = """require ["fileinto"];
+
+# rule:[test]
+if anyof (exists ["Subject"]) {
+    fileinto "INBOX";
+}
+"""
+        fs = self._load(script)
+        conds = fs.get_filter_conditions("[test]")
+        self.assertEqual(len(conds), 1)
+        self.assertEqual(conds[0][0], "exists")
+        self.assertIn("Subject", conds[0])
+
+    def test_exists_multiple_headers(self):
+        script = """require ["fileinto"];
+
+# rule:[test]
+if anyof (exists ["Subject", "From"]) {
+    fileinto "INBOX";
+}
+"""
+        fs = self._load(script)
+        conds = fs.get_filter_conditions("[test]")
+        self.assertEqual(len(conds), 1)
+        self.assertEqual(conds[0][0], "exists")
+        self.assertIn("Subject", conds[0])
+        self.assertIn("From", conds[0])
+
+    def test_exists_plain_string(self):
+        script = """require ["fileinto"];
+
+# rule:[test]
+if exists "Subject" {
+    fileinto "INBOX";
+}
+"""
+        fs = self._load(script)
+        conds = fs.get_filter_conditions("[test]")
+        self.assertEqual(conds[0][0], "exists")
+        self.assertIn("Subject", conds[0])
