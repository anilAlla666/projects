I've added a new test file that checks fragment-only links are not treated as relative links. I haven't run it against the current code or a fix. The tests call `ConfluenceRenderer.link` directly and also go through `mistune.Markdown`, which assumes the mistune 2.x API this renderer is built on.

The tests assert only that `#fragment` links are not recorded in `relative_links` and that no `md2cf-internal-link` placeholder appears in the output. They do not assert the exact output `href`, since the issue leaves that open. Relative links to other pages, such as `other.md#section`, must still be recorded.

```diff

+import mistune
+
+from md2cf.confluence_renderer import ConfluenceRenderer
+
+
+def test_fragment_only_link_is_not_relative_link():
+    renderer = ConfluenceRenderer(enable_relative_links=True)
+    output = renderer.link("#frequently-asked-questions-osep", None, "FAQ")
+
+    assert renderer.relative_links == []
+    assert "md2cf-internal-link" not in output
+    assert "FAQ" in output
+
+
+def test_fragment_only_link_via_markdown_is_not_relative_link():
+    renderer = ConfluenceRenderer(enable_relative_links=True)
+    markdown = mistune.Markdown(renderer=renderer)
+    output = markdown(
+        "[Frequently Asked Questions](#frequently-asked-questions-osep)\n"
+    )
+
+    assert renderer.relative_links == []
+    assert "md2cf-internal-link" not in output
+
+
+def test_relative_link_with_path_still_detected():
+    renderer = ConfluenceRenderer(enable_relative_links=True)
+    output = renderer.link("other.md", None, "other")
+
+    assert len(renderer.relative_links) == 1
+    assert renderer.relative_links[0].path == "other.md"
+    assert renderer.relative_links[0].replacement in output
+
+
+def test_relative_link_with_path_and_fragment_still_detected():
+    renderer = ConfluenceRenderer(enable_relative_links=True)
+    renderer.link("other.md#section", None, "other")
+
+    assert len(renderer.relative_links) == 1
+    assert renderer.relative_links[0].path == "other.md"
+    assert renderer.relative_links[0].fragment == "section"
+
+
+def test_external_link_not_relative():
+    renderer = ConfluenceRenderer(enable_relative_links=True)
+    renderer.link("https://example.com/#foo", None, "ext")
+
+    assert renderer.relative_links == []
```
