+def test_srun_env_var(monkeypatch):
+    monkeypatch.setenv('SLURM_CPUS_PER_TASK', '2')
+    monkeypatch.setenv('SRUN_CPUS_PER_TASK', '')
+
+    launcher = getlauncher('srun')()
+    job = Job('host', 'user', 'fake_job', 'fake_stdout', 'fake_stderr',
+              1, 1, 1, 1, 1, 1, 1, 1, 1, {'foo': None})
+
+    command = launcher.run_command(job)
+    assert command == 'srun --cpus-per-task=2 --foo'
+
+    # Test that SRUN_CPUS_PER_TASK takes precedence over SLURM_CPUS_PER_TASK
+    monkeypatch.setenv('SRUN_CPUS_PER_TASK', '4')
+    command = launcher.run_command(job)
+    assert command == 'srun --cpus-per-task=4 --foo'
+
+
