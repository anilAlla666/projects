+import pytest
+from markdownify import markdownify as md
+
+
+def test_figure_and_table_captions():
+    assert md('TEXT<table><caption>Caption</caption><tr><td>CELL</td></tr></tbody></table>') == 'TEXT\n\nCaption\n| CELL |\n\n'
+    assert md('TEXT<figure><figcaption>Caption</figcaption><span>SPAN</span></figure>') == 'TEXT\n\nCaption\nSPAN\n'
+    assert md('<figure><span>SPAN</span><figcaption>Caption</figcaption></figure>TEXT') == 'SPAN\n\nCaption\nTEXT'
+
+
+def test_blank_lines_around_list_items():
+    assert md('<ul><li>Item 1</li><li>Item 2</li></ul>') == '- Item 1\n- Item 2\n'
+    assert md('<ol><li>Item 1</li><li>Item 2</li></ol>') == '1. Item 1\n2. Item 2\n'
+    assert md('<dl><dt>Title 1</dt><dd>Description 1</dd><dt>Title 2</dt><dd>Description 2</dd></dl>') == '> Title 1\nDescription 1\n> Title 2\nDescription 2\n'
