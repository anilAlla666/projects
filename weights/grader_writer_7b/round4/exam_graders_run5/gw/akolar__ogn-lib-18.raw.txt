+def get_messages(n_messages=float('inf')):
+    test_dir = os.path.dirname(os.path.realpath(__file__))
+
+    with open(os.path.join(test_dir, 'messages.txt'), 'r') as f:
+        lines = f.readlines()
+
+    messages = []
+    for i, l in enumerate(lines):
+        if i >= n_messages:
+            break
+
+        messages.append(l.strip())
+
+    return messages
