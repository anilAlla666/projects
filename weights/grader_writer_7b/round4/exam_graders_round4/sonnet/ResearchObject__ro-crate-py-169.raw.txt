def _try_set_id(entity, value):

def test_dataset_id_not_modifiable():
    crate = ROCrate()
    d = crate.add_dataset("FOO")
    old_id = d.id
    _try_set_id(d, "foo")
    assert d.id == old_id
    assert crate.dereference("FOO") is d
    assert crate.dereference("foo") is None


def test_contextual_entity_id_not_modifiable():
    crate = ROCrate()
    joe = crate.add(Person(crate, "#joe", properties={"name": "Joe"}))
    _try_set_id(joe, "#jim")
    assert joe.id == "#joe"
    assert crate.get("#joe") is joe
    assert crate.get("#jim") is None


def test_id_assignment_raises():
    crate = ROCrate()
    f = crate.add_dataset("bar")
