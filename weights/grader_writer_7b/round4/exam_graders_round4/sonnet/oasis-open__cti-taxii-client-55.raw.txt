I've added a test file covering `Collection` URLs with and without a trailing slash. I haven't run it against the current code or a fix.

The tests use the `responses` mock library. They register the collection endpoint with and without a trailing slash, so they don't depend on whether a fix normalizes `self.url`. The objects endpoints are registered only at the correct, slash-separated URLs. Before the fix, `get_object` and `get_objects` on a slash-less collection URL request the glued path (`...<id>objects/...`), so `responses` raises a connection error and the tests fail. They also check that no request URL contains the glued `<id>objects` path. The slash-terminated cases should already pass and guard against regressions.

I assumed the collection endpoint returns the usual JSON fields (`id`, `title`, `can_read`, `can_write`, `media_types`) and that `get_object` and `get_objects` return parsed JSON.

+import json
+
+import responses
+
+from taxii2client import Collection
+
+HOST = "http://localhost:5000/trustgroup1/collections/"
+COLL_ID = "91a7b528-80eb-42ed-a74d-c6fbd5a26116"
+BASE = HOST + COLL_ID
+OBJ_ID = "indicator--252c7c11-daf2-42bd-843b-be65edca9f61"
+
+TAXII = "application/vnd.oasis.taxii+json; version=2.0"
+STIX = "application/vnd.oasis.stix+json; version=2.0"
+
+COLLECTION_INFO = {
+    "id": COLL_ID,
+    "title": "Test collection",
+    "description": "desc",
+    "can_read": True,
+    "can_write": True,
+    "media_types": [STIX],
+}
+
+OBJECT = {
+    "type": "bundle",
+    "id": "bundle--8f4d2c1e-5a4b-4c0a-9c58-3d0f6b6e2a11",
+    "spec_version": "2.0",
+    "objects": [],
+}
+
+
+def _register_collection():
+    for u in (BASE, BASE + "/"):
+        responses.add(responses.GET, u, body=json.dumps(COLLECTION_INFO),
+                      status=200, content_type=TAXII)
+
+
+def _requested_urls():
+    return [c.request.url for c in responses.calls]
+
+
+@responses.activate
+def test_get_object_no_trailing_slash():
+    _register_collection()
+    responses.add(responses.GET, BASE + "/objects/" + OBJ_ID + "/",
+                  body=json.dumps(OBJECT), status=200, content_type=STIX)
+
+    coll = Collection(BASE, user="user1", password="Password1")
+    coll.get_object(OBJ_ID)
+
+    for url in _requested_urls():
+        assert COLL_ID + "objects" not in url
+    assert any(u.split("?")[0] == BASE + "/objects/" + OBJ_ID + "/"
+               for u in _requested_urls())
+
+
+@responses.activate
+def test_get_object_with_trailing_slash():
+    _register_collection()
+    responses.add(responses.GET, BASE + "/objects/" + OBJ_ID + "/",
+                  body=json.dumps(OBJECT), status=200, content_type=STIX)
+
+    coll = Collection(BASE + "/", user="user1", password="Password1")
+    coll.get_object(OBJ_ID)
+
+    assert any(u.split("?")[0] == BASE + "/objects/" + OBJ_ID + "/"
+               for u in _requested_urls())
+
+
+@responses.activate
+def test_get_objects_no_trailing_slash():

+                  body=json.dumps(OBJECT), status=200, content_type=STIX)
+
+    coll = Collection(BASE, user="user1", password="Password1")
+    coll.get_objects()
+
+    for url in _requested_urls():
+        assert COLL_ID + "objects" not in url
+    assert any(u.split("?")[0] == BASE + "/objects/"
+               for u in _requested_urls())
+
