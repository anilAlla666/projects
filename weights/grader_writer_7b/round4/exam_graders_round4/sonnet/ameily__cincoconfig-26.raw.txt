The diff below adds one test file. It fails on the current code because `to_basic('')` encrypts the empty string, and it should pass once empty strings are treated like `None`. I have not run it.

+from unittest.mock import MagicMock
+
+from cincoconfig.fields import SecureField
+
+
+def _make_cfg():
+    cfg = MagicMock()
+    ctx = MagicMock()
+    secret = MagicMock()
+    secret.method = 'aes'
+    secret.ciphertext = b'abcd'
+    ctx.encrypt.return_value = secret
+    cfg._keyfile.__enter__.return_value = ctx
+    return cfg, ctx
+
+
+class TestSecureFieldEmptyString:
+
+    def test_to_basic_empty_string_is_none(self):
+        cfg, ctx = _make_cfg()
+        field = SecureField(name='secret')
+        assert field.to_basic(cfg, '') is None
+
+    def test_to_basic_empty_string_not_encrypted(self):
+        cfg, ctx = _make_cfg()
+        field = SecureField(name='secret')
+        field.to_basic(cfg, '')
+        ctx.encrypt.assert_not_called()
+
+    def test_to_basic_nonempty_still_encrypted(self):
+        cfg, ctx = _make_cfg()
+        field = SecureField(name='secret')
