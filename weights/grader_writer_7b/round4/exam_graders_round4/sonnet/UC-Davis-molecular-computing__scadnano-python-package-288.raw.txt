I added a new test file covering both crashing examples from the issue. I did not run the tests. They assume the fix keeps the existing delimiter behaviour, with the delimiter placed between the DNA of each domain.

The tests check the exported DNA rather than strand names or the other bulk-format columns, so they don't depend on details the issue leaves open. Each case checks `vendor_dna_sequence()` with the default and a `_` delimiter, and checks that `to_idt_bulk_input_format()` runs and includes the full sequence. A fifth test combines an extension and a loopout in one strand.

+import unittest
+
+import scadnano as sc
+
+
+class TestVendorExportExtensionsLoopouts(unittest.TestCase):
+
+    def _extension_design(self) -> sc.Design:
+        design = sc.Design(helices=[sc.Helix(max_offset=100)])
+        design.draw_strand(0, 0).extension_5p(3).move(5).with_sequence('TTT' + 'AAAAA')
+        return design
+
+    def _loopout_design(self) -> sc.Design:
+        design = sc.Design(helices=[sc.Helix(max_offset=100), sc.Helix(max_offset=100)])
+        design.draw_strand(0, 0).move(5).loopout(1, 3).move(-5).with_sequence('AAAAA' + 'TTT' + 'AAAAA')
+        return design
+
+    def test_extension_vendor_dna_sequence(self) -> None:
+        design = self._extension_design()
+        strand = design.strands[0]
+        self.assertEqual('TTTAAAAA', strand.vendor_dna_sequence())
+
+    def test_extension_vendor_dna_sequence_delimiter(self) -> None:
+        design = self._extension_design()
+        strand = design.strands[0]
+        self.assertEqual('TTT_AAAAA', strand.vendor_dna_sequence(domain_delimiter='_'))
+
+    def test_extension_idt_bulk_input(self) -> None:
+        design = self._extension_design()
+        contents = design.to_idt_bulk_input_format()
+        self.assertIn('TTTAAAAA', contents)
+
+    def test_loopout_vendor_dna_sequence(self) -> None:
+        design = self._loopout_design()
+        strand = design.strands[0]
+        self.assertEqual('AAAAATTTAAAAA', strand.vendor_dna_sequence())
+
+    def test_loopout_vendor_dna_sequence_delimiter(self) -> None:
+        design = self._loopout_design()
+        strand = design.strands[0]
+        self.assertEqual('AAAAA_TTT_AAAAA', strand.vendor_dna_sequence(domain_delimiter='_'))
+
+    def test_loopout_idt_bulk_input(self) -> None:
+        design = self._loopout_design()
+        contents = design.to_idt_bulk_input_format()
+        self.assertIn('AAAAATTTAAAAA', contents)
+
+    def test_extension_and_loopout_3p(self) -> None:
+        design = sc.Design(helices=[sc.Helix(max_offset=100), sc.Helix(max_offset=100)])
+        design.draw_strand(0, 0).extension_5p(2).move(5).loopout(1, 3).move(-5) \
