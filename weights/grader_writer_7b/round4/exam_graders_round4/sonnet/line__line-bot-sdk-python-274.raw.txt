- **Through `FlexSendMessage`:** the box hero from the issue's script, passed as `contents`, must serialize with `type: box` and no `url`.

+import unittest
+
+from linebot.models import (
+    FlexSendMessage,
+    BubbleContainer,
+    BoxComponent,
+    ImageComponent,

+class TestBubbleHero(unittest.TestCase):
+    BOX_HERO = {
+        'type': 'box',
+        'layout': 'vertical',
+        'contents': [
+            {'type': 'text', 'text': 'Brown Cafe'}
+        ]
+    }
+
+    IMAGE_HERO = {
+        'type': 'image',
+        'url': 'https://example.com/image.png',
+        'size': 'full'
+    }
+
+    def test_hero_box_from_dict(self):
+        bubble = BubbleContainer(hero=self.BOX_HERO)
+        self.assertIsInstance(bubble.hero, BoxComponent)
+        self.assertEqual(bubble.hero.layout, 'vertical')
+        self.assertEqual(len(bubble.hero.contents), 1)
+        self.assertIsInstance(bubble.hero.contents[0], TextComponent)
+
+        hero_json = bubble.as_json_dict()['hero']
+        self.assertEqual(hero_json['type'], 'box')
+        self.assertNotIn('url', hero_json)
+        self.assertEqual(hero_json['layout'], 'vertical')
+        self.assertEqual(hero_json['contents'][0]['text'], 'Brown Cafe')
+
+    def test_hero_image_from_dict(self):
+        bubble = BubbleContainer(hero=self.IMAGE_HERO)
+        self.assertIsInstance(bubble.hero, ImageComponent)
+        hero_json = bubble.as_json_dict()['hero']
+        self.assertEqual(hero_json['type'], 'image')
+        self.assertEqual(hero_json['url'], 'https://example.com/image.png')
+
+    def test_hero_component_objects(self):
+        box = BoxComponent(layout='vertical',
+                           contents=[TextComponent(text='hi')])
+        bubble = BubbleContainer(hero=box)
+        self.assertIsInstance(bubble.hero, BoxComponent)
+        self.assertEqual(bubble.as_json_dict()['hero']['type'], 'box')
+
+        image = ImageComponent(url='https://example.com/a.png')
+        bubble = BubbleContainer(hero=image)
+        self.assertIsInstance(bubble.hero, ImageComponent)
+        self.assertEqual(bubble.as_json_dict()['hero']['type'], 'image')
+
+    def test_flex_send_message_with_box_hero(self):
+        contents = {
+            'type': 'bubble',
+            'hero': self.BOX_HERO,
+        }
+        message = FlexSendMessage(alt_text='altText', contents=contents)
+        self.assertIsInstance(message.contents, BubbleContainer)
+        self.assertIsInstance(message.contents.hero, BoxComponent)
+
+        hero_json = message.as_json_dict()['contents']['hero']
+        self.assertEqual(hero_json['type'], 'box')
+        self.assertNotIn('url', hero_json)
+        self.assertEqual(hero_json['contents'][0]['type'], 'text')
