+from fffw.graph import VIDEO, meta

+class TrimDurationTestCase(TestCase):

+
+    def setUp(self) -> None:

+    def test_trim_end_after_eof(self):
+        """ Trim end after EOF must not extend stream duration."""
+        trim = filters.Trim(VIDEO, start=2.0, end=20.0)
+        result = trim.transform(self.meta)
