The two tests below call `ng.peakpick.pick` on 2D data with `algorithm="thres"` and `"thres-fast"`. They have not been run against the current code or a fix. The failing path is the one the issue describes: `est_params=True` indexes `data` with a list of slices, which numpy 1.22 and later reject. The tests only assert that two synthetic Gaussian peaks are found, so they do not depend on how the fix is made (`tuple(seg_slice)` or a changed return type). I also did not rely on the exact output layout of `pick` beyond it having one row per peak.

+"""Tests for ng.peakpick.pick with the 'thres' and 'thres-fast' algorithms."""
+import numpy as np
+import pytest
+
+import nmrglue as ng
+
+
+def _make_data():
+    y, x = np.mgrid[0:64, 0:64]
+    data = np.zeros((64, 64))
+    for (cy, cx, amp) in [(20, 20, 100.0), (44, 40, 80.0)]:
+        data += amp * np.exp(-((y - cy) ** 2 + (x - cx) ** 2) / (2 * 3.0 ** 2))
+    return data
+
+
+@pytest.mark.parametrize("algorithm", ["thres", "thres-fast"])
+def test_pick_thres_est_params(algorithm):
+    data = _make_data()
+    peaks = ng.peakpick.pick(
+        data, pthres=10.0, msep=(3, 3), algorithm=algorithm,
+        est_params=True, cluster=False, table=True)
+    assert len(peaks) == 2
+    found = sorted(
+        (int(round(p[0])), int(round(p[1]))) for p in
+        [tuple(np.atleast_1d(list(r))[:2]) for r in peaks])
+    assert found == [(20, 20), (44, 40)]
+
+
+@pytest.mark.parametrize("algorithm", ["thres", "thres-fast"])
+def test_pick_thres_cluster(algorithm):
+    data = _make_data()
+    peaks = ng.peakpick.pick(
+        data, pthres=10.0, msep=(3, 3), algorithm=algorithm,
+        est_params=True, cluster=True, table=True)
+    assert len(peaks) == 2
