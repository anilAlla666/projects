I've added one test file covering the table caption and both figcaption cases from the issue. I haven't run it. The tests use regular expressions that require at least one blank line (`\n\n` or more) between the caption and its neighbours. They don't depend on the exact number of newlines the fix emits.

The table test requires a blank line between the caption and the first `|`, which matches the issue title ("blank lines are missing around … table captions"). A fix that only inserted a single `\n` would fail it.

+import re
+
+from markdownify import markdownify as md
+
+
+def test_table_caption_separated_from_table():
+    out = md('TEXT<table><caption>Caption</caption><tr><td>CELL</td></tr></tbody></table>')
+    assert re.search(r'TEXT\n\n+Caption\n\n+\| CELL \|', out), repr(out)
+
+
+def test_figcaption_at_top_of_figure():
+    out = md('TEXT<figure><figcaption>Caption</figcaption><span>SPAN</span></figure>')
+    assert re.search(r'TEXT\n\n+Caption\n\n+SPAN', out), repr(out)
+
+
+def test_figcaption_at_bottom_of_figure():
+    out = md('<figure><span>SPAN</span><figcaption>Caption</figcaption></figure>TEXT')
+    assert re.search(r'SPAN\n\n+Caption\n\n+TEXT', out), repr(out)
