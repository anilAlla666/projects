I haven't run these tests. They rest on one assumption I couldn't check, because the shown part of `dark/reads.py` doesn't include the reverse-complement code. I believe `DNARead.reverseComplement` builds its complement table from Biopython's `ambiguous_dna_complement`, which only has uppercase keys. If so, lowercase bases are left uncomplemented on the reverse strand, which would corrupt the three reverse-strand translations at lowercase sites.

The tests compare results case-insensitively, so they don't dictate whether a fix returns lowercase or uppercase complements. They cover:
- the reverse complement of lowercase and mixed-case DNA and RNA;
- lowercase and mixed-case translations matching the uppercase ones, in all six frames.

+from unittest import TestCase
+
+from dark.reads import DNARead, RNARead
+
+
+class TestLowercaseReverseComplement(TestCase):
+    """
+    Reverse complementing and translating lowercase sequences must give
+    the same result as for uppercase sequences (ignoring output case).
+    """
+
+    def testDNAReverseComplementLowercase(self):
+        read = DNARead('id', 'aacc')
+        self.assertEqual('GGTT', read.reverseComplement().sequence.upper())
+
+    def testDNAReverseComplementMixedCase(self):
+        read = DNARead('id', 'aAcCgT')
+        self.assertEqual('ACGGTT',

+
+    def testRNAReverseComplementLowercase(self):
+        read = RNARead('id', 'aacc')
+        self.assertEqual('GGUU', read.reverseComplement().sequence.upper())
+
+    def testLowercaseTranslationsMatchUppercase(self):
+        seq = 'atggccaaacgtttagctgattgcaagtcc'
+        upper = [(t.sequence.upper(), t.frame, t.reverseComplemented)
+                 for t in DNARead('id', seq.upper()).translations()]
+        lower = [(t.sequence.upper(), t.frame, t.reverseComplemented)
+                 for t in DNARead('id', seq).translations()]
+        self.assertEqual(upper, lower)
+
+    def testMixedCaseTranslationsMatchUppercase(self):
+        seq = 'ATGgccAAAcgtTTAgctGATtgcAAGtcc'
+        upper = [(t.sequence.upper(), t.frame, t.reverseComplemented)
+                 for t in DNARead('id', seq.upper()).translations()]
+        mixed = [(t.sequence.upper(), t.frame, t.reverseComplemented)
+                 for t in DNARead('id', seq).translations()]
+        self.assertEqual(upper, mixed)

+        # Reverse complement of 'cccatt' is 'aatggg' -> N G
