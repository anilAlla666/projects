+def test_scan_iter_types(r: redis.Redis) -> None:
+    r.set("str", "val")
+    r.lpush("list", "val")
+    r.sadd("set", "val")
+    r.zadd("zset", {"val": 1})
+    r.hset("hash", "field", "val")
+    assert r.scan_iter(_type="STRING") == [b"str"]
+    assert r.scan_iter(_type="LIST") == [b"list"]
+    assert r.scan_iter(_type="SET") == [b"set"]
+    assert r.scan_iter(_type="ZSET") == [b"zset"]
+    assert r.scan_iter(_type="HASH") == [b"hash"]
+    assert r.scan_iter(_type="NONEXISTENTTYPE") == []
