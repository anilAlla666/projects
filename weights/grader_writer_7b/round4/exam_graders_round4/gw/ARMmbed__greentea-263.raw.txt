+import unittest
+from mbed_greentea.mbed_greentea_cli import create_filtered_test_list
+
+
+class TestCreateFilteredTestList(unittest.TestCase):
+    def setUp(self):
+        self.ctest_test_list = {
+            'TESTS-mbed_drivers': ['TESTS-mbed_drivers-rtc', 'TESTS-mbed_drivers-ticker'],
+            'TESTS-mbed_hal': ['TESTS-mbed_hal-lp_ticker', 'TESTS-mbed_hal-lp_ticker2'],
+            'TESTS-mbed_overrides': ['TESTS-mbed_overrides-rtc', 'TESTS-mbed_overrides-ticker']
+        }
+
+    def test_filter_names_by_prefix(self):
+        expected_result = [
+            'TESTS-mbed_drivers-rtc',
+            'TESTS-mbed_drivers-ticker',
+            'TESTS-mbed_hal-lp_ticker',
+            'TESTS-mbed_hal-lp_ticker2',
+            'TESTS-mbed_overrides-rtc',
+            'TESTS-mbed_overrides-ticker'
+        ]
+        result = create_filtered_test_list(
+            self.ctest_test_list,
+            'TESTS-*',
+            ''
+        )
+        self.assertEqual(expected_result, result.keys())
+
+    def test_filter_names_by_prefix_star_suffix(self):
+        expected_result = [
+            'TESTS-mbed_drivers-rtc',
+            'TESTS-mbed_drivers-ticker',
+            'TESTS-mbed_hal-lp_ticker',
+            'TESTS-mbed_hal-lp_ticker2',
+            'TESTS-mbed_overrides-rtc',
+            'TESTS-mbed_overrides-ticker'
+        ]
+        result = create_filtered_test_list(
+            self.ctest_test_list,
+            '*-ticker',
+            ''
+        )
+        self.assertEqual(expected_result, result.keys())
+
+    def test_filter_names_by_prefix_star_suffix_2(self):
+        expected_result = [
+            'TESTS-mbed_drivers-rtc',
+            'TESTS-mbed_drivers-ticker',
+            'TESTS-mbed_hal-lp_ticker',
+            'TESTS-mbed_hal-lp_ticker2',
+            'TESTS-mbed_overrides-rtc',
+            'TESTS-mbed_overrides-ticker'
+        ]
+        result = create_filtered_test_list(
+            self.ctest_test_list,
+            'TESTS-*-ticker',
+            ''
+        )
+        self.assertEqual(expected_result, result.keys())
+
+    def test_filter_names_by_prefix_star_suffix_3(self):
+        expected_result = [
+            'TESTS-mbed_drivers-rtc',
+            'TESTS-mbed_drivers-ticker',
+            'TESTS-mbed_hal-lp_ticker',
+            'TESTS-mbed_hal-lp_ticker2',
+            'TESTS-mbed_overrides-rtc',
+            'TESTS-mbed_overrides-ticker'
+        ]
+        result = create_filtered_test_list(
+            self.ctest_test_list,
+            'TESTS-*',
+            '',
+            True
+        )
+        self.assertEqual(expected_result, result.keys())
