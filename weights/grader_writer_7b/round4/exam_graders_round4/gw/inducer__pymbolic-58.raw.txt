+class FlopCounter(IdentityMapper):
+    def map_sum(self, expr, **kwargs):
+        return sum(map(self.rec, expr.operands, **kwargs)) + 1
+
+    def map_product(self, expr, **kwargs):
+        return sum(map(self.rec, expr.operands, **kwargs)) + 1
+
+    def map_quotient(self, expr, **kwargs):
+        return sum(map(self.rec, expr.operands, **kwargs)) + 1
+
+    def map_power(self, expr, **kwargs):
+        return sum(map(self.rec, expr.operands, **kwargs)) + 1
+
+    def map_call(self, expr, **kwargs):
+        return sum(map(self.rec, expr.parameters, **kwargs)) + 1
+
+
+class CSEAwareFlopCounter(FlopCounter):
+    def __init__(self):
+        super().__init__()
+        self._seen = set()
+
+    def rec(self, expr, **kwargs):
+        if expr in self._seen:
+            return 0
+        else:
+            self._seen.add(expr)
+            return super().rec(expr, **kwargs)
+

+def test_if_else_precedence():
+    assert parse("10 if x or y else 20") == \
+           prim.IfElse(prim.Or(prim.Variable("x"), prim.Variable("y")),
+                       prim.Literal(10),
+                       prim.Literal(20))
