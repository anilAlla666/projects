+    def test_verify_with_byte_strings(self):
+        secret = b'secret'
+        data = {'parĄm1': b'value1'}
+        base_url = b'https://endpoint.com/ąpi'
+        method = 'POST'
+
+        signature = signing.sign(secret, data, base_url, method)
+
+        self.assertTrue(signing.verify(signature, secret, data, base_url, method))
+
+        # Changing the value in the params
+        false_data = {'parĄm1': b'value2'}
+        self.assertFalse(signing.verify(signature, secret, false_data, base_url, method))
