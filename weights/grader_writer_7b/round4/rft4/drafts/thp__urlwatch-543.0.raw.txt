+class TestShellPipeFilter:
+    def setup_method(self):
+        self.state = MockJobState("mock-job-name", "mock-location")
+
+    @pytest.mark.usefixtures("monkeypatch_env_clear")
+    def test_shell_pipe_with_environment_variables_added(self, monkeypatch_env):
+        """
+        Verifies that the `environment_variables` property of the `MockJobState`
+        object gets merged into the environment before executing the shell pipe.
+        """
+        monkeypatch_env["SOME_ENV_VAR"] = "some value"
+        monkeypatch_env["ANOTHER_ENV_VAR"] = "another value"
+        monkeypatch_env["A_THIRD_ENV_VAR"] = "a third value"
+
+        expected_env_vars = {
+            "SOME_ENV_VAR": "some value",
+            "ANOTHER_ENV_VAR": "another value",
+            "A_THIRD_ENV_VAR": "a third value",
+            "URLWATCH_JOB_NAME": "mock-job-name",
+            "URLWATCH_JOB_LOCATION": "mock-location",
+        }
+
+        shell_pipe_filter = FilterBase.filter_class_for("shellpipe")(self.state, None)
+        actual_env_vars = shell_pipe_filter.environment_variables
+
+        assert actual_env_vars == expected_env_vars
+
+    @pytest.mark.usefixtures("monkeypatch_env")
+    def test_shell_pipe_with_environment_variables_overwritten(self, monkeypatch_env):
+        """
+        Verifies that the `environment_variables` property of the `MockJobState`
+        object overwrites any pre-existing environment variables.
+        """
+        monkeypatch_env["URLWATCH_JOB_NAME"] = "overridden job name"
+        monkeypatch_env["URLWATCH_JOB_LOCATION"] = "overridden location"
+
+        expected_env_vars = {
+            "URLWATCH_JOB_NAME": "mock-job-name",
+            "URLWATCH_JOB_LOCATION": "mock-location",
+        }
+
+        shell_pipe_filter = FilterBase.filter_class_for("shellpipe")(self.state, None)
+        actual_env_vars = shell_pipe_filter.environment_variables
+
+        assert actual_env_vars == expected_env_vars
