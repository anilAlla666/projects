+    partition = TotalConnectPartition(PARTITION_DETAILS_1, None)
+    partition._update(PARTITION_DISARMED)
+    assert partition.arming_state.value == PARTITION_DISARMED["ArmingState"]
+
+    # now check with unknown state
+    partition = TotalConnectPartition(PARTITION_DETAILS_1, None)
+    partition._update({"ArmingState": 999999999})
+    assert partition.arming_state.value == 999999999
+
+    # check with None
+    partition = TotalConnectPartition(PARTITION_DETAILS_1, None)
+    with pytest.raises(PartialResponseError):
+        partition._update({})
+
+    # check with empty
+    partition = TotalConnectPartition(PARTITION_DETAILS_1, None)
+    partition._update({"ArmingState": ""})
+    assert partition.arming_state.value == ""
+
+    # check with zero
+    partition = TotalConnectPartition(PARTITION_DETAILS_1, None)
+    partition._update({"ArmingState": 0})
+    assert partition.arming_state.value == 0
+
+    # check with non-integer
+    partition = TotalConnectPartition(PARTITION_DETAILS_1, None)
+    partition._update({"ArmingState": "not_an_int"})
+    assert partition.arming_state.value == "not_an_int"
+
+    # check with negative
+    partition = TotalConnectPartition(PARTITION_DETAILS_1, None)
+    partition._update({"ArmingState": -1})
