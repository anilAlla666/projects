+import json
+from sendgrid.helpers.mail import (Email)
+
+
+try:
+    import unittest2 as unittest
+except ImportError:
+    import unittest
+
+
+class TestEmailObject(unittest.TestCase):
+    def test_from_str(self):
+        name = "SomeName"
+        address = "test@example.com"
+        email = Email(name_address="{0} <{1}>".format(name, address))
+        self.assertEqual(email.name, name)
+        self.assertEqual(email.email, "test@example.com")
+
+    def test_from_dict(self):
+        name = "SomeName"
+        address = "test@example.com"
+        email = Email(name_address={"name": name, "email": address})
+        self.assertEqual(email.name, name)
+        self.assertEqual(email.email, "test@example.com")
+
+    def test_from_str_and_dict(self):
+        name = "SomeName"
+        address = "test@example.com"
+        email = Email(name_address=[{"name": name, "email": address}, name])
+        self.assertEqual(email.name, name)
+        self.assertEqual(email.email, "test@example.com")
