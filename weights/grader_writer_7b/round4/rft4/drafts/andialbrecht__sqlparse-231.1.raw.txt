+def test_comparison_with_function():
+    p = sqlparse.parse('foo = date(bar.baz)')[0].tokens[0]
+    assert p.get_type() == 'Comparison'
+    comparison = p
+    assert comparison.left.tokens[0].get_real_name() == 'foo'
+    assert comparison.right.get_type() == 'Function'
+    function = comparison.right
+    assert function.name == 'date'
+    assert function.tokens[0].get_real_name() == 'bar.baz'
+    assert comparison.operator == '='
