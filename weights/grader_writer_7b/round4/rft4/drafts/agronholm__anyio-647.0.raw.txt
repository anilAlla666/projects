+def test_backend_options(name, expected, backend):
+    async def main():
+        assert isinstance(anyio.get_current_task(), anyio._backends._base.Task)
+        assert isinstance(anyio.get_running_backend(), anyio.Backend)
+
+    options = {
+        f"default_tcp_socket_{key}": value
+        for key, value in {"family", "type", "proto"}.intersection(
+            name.split("_")[1:]
+        )
+    }
+    if name.startswith("use_"):
+        options[name] = True
+    elif name.startswith("trio_lowlevel_"):
+        options[name] = True
+    elif name.startswith("uvloop"):
+        options[name] = False
+    else:
+        pytest.fail(f"Unexpected option: {name}")
+
+    with patch.dict("anyio._backends.__dict__", {f"options_{backend}": options}):
+        anyio.run(main, backend=backend)
