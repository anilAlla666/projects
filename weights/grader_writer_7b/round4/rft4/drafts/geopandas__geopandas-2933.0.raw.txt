+def test_merge_retains_gdf():
+    # https://github.com/geopandas/geopandas/issues/2931
+    point = shapely.Point(600000, 5000000)
+    df_1 = geopandas.GeoDataFrame(data={"geom": [shapely.Point([1, 2])]}, geometry="geom", crs="25832")
+    df_merged = df_1.merge(df_1, left_index=True, right_index=True)
+    assert type(df_merged) == geopandas.GeoDataFrame
+    res = pd.concat([df_1.geom.rename("geom_x"), df_1.geom.rename("geom_y")], axis=1)
+    assert type(res) == geopandas.GeoDataFrame
