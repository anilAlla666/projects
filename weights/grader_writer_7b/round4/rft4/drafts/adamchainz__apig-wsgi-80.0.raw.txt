+import sys
+from base64 import b64encode
+from io import BytesIO
+import pytest
+from apig_wsgi import make_lambda_handler
+
+
+@pytest.fixture()
+def simple_app():
+    def app(environ, start_response):
+        app.environ = environ
+        start_response("200 OK", app.headers, app.exc_info)
+        return BytesIO(app.response)
+
+    app.headers = [("Content-Type", "text/plain")]
+    app.response = b"Hello World\n"
+    app.handler = make_lambda_handler(app)
+    app.exc_info = None
+    yield app

+def make_event(
+    method="GET",
+    qs_params=None,
+    headers=None,
+    body="",
+    binary=False,
+    request_context=None,
+):
+    if headers is None:
+        headers = {"Host": "example.com"}
+
+    event = {
+        "httpMethod": method,
+        "path": "/",
+        "queryStringParameters": qs_params,
+        "headers": headers,
+    }
+    if binary:
+        event["body"] = b64encode(body.encode("utf-8")).decode("utf-8")
+        event["isBase64Encoded"] = True
+    else:
+        event["body"] = body
+
+    if request_context is not None:
+        event["requestContext"] = request_context
+
+    return event
+
+
+class TestResponse:
+    def test_binary_support(self, simple_app):
+        event = make_event(headers={"Content-Type": "application/json"})
+        resp = simple_app.handler(event, None)["headers"]
+        assert resp["x-apig-wsgi-binary-support"] == "True"
+
+    def test_non_binary_support(self, simple_app):
+        event = make_event(headers={"Content-Type": "text/plain"})
+        resp = simple_app.handler(event, None)["headers"]
+        assert resp["x-apig-wsgi-binary-support"] == "False"
+
+    def test_gzip_encoded(self, simple_app):
+        event = make_event(headers={"Content-Type": "text/plain"}, body="a"*2048)
+        resp = simple_app.handler(event, None)["headers"]
+        assert resp["x-apig-wsgi-binary-support"] == "True"
+
+    def test_binary_not_supported(self, simple_app):
+        event = make_event(headers={"Content-Type": "application/json"})
+        resp = simple_app.handler(event, None)["headers"]
+        assert resp["x-apig-wsgi-binary-support"] == "False"
+
+    def test_binary_supported(self, simple_app):
+        event = make_event(
+            headers={"Content-Type": "application/json"},
+            body=b"\x1f\x8b\x08\x00\x00\x00\x00\x00\x00\xff",
+        )
+        resp = simple_app.handler(event, None)["headers"]
+        assert resp["x-apig-wsgi-binary-support"] == "True"
