+class DataUriTest(unittest.TestCase):
+    def test_empty_data_uri(self):
+        result = parse_data_uri("")
+        self.assertIsNone(result)
+
+    def test_no_scheme(self):
+        result = parse_data_uri("A%20brief%20note")
+        self.assertEqual(result.data, b"A brief note")
+
+    def test_bare_media_type(self):
+        result = parse_data_uri("text/plain,A%20brief%20note")
+        self.assertEqual(result.media_type, "text/plain")
+        self.assertEqual(result.data, b"A brief note")
+
+    def test_bare_media_type_and_params(self):
+        result = parse_data_uri("text/plain;charset=UTF-8,A%20brief%20note")
+        self.assertEqual(result.media_type, "text/plain")
+        self.assertEqual(result.media_type_parameters,
+                         {"charset": "UTF-8"})
+        self.assertEqual(result.data, b"A brief note")
+
+    def test_bare_media_type_and_charset(self):
+        result = parse_data_uri("text/plain; charset=UTF-8,A%20brief%20note")
+        self.assertEqual(result.media_type, "text/plain")
+        self.assertEqual(result.media_type_parameters,
+                         {"charset": "UTF-8"})
+        self.assertEqual(result.data, b"A brief note")
+
