+def assert_bqm_equal(this, that):
+    this_adj = this.adj
+    that_adj = that.adj
+    this_linear = this.linear
+    that_linear = that.linear
+    this_quadratic = this.quadratic
+    that_quadratic = that.quadratic
+    this_offset = this.offset
+    that_offset = that.offset
+
+    assert set(this_adj.keys()) == set(that_adj.keys()), "this variables != that variables"
+
+    assert len(this_adj) == len(that_adj), "this variables != that variables"
+
+    for u in this_adj:
+        assert len(this_adj[u]) == len(that_adj[u]), "adjacency lengths don't match"
+        for v in this_adj[u]:
+            assert u in that_adj[v], "undirected graph check failed"
+            assert v in that_adj[u], "undirected graph check failed"
+            assert this_adj[u][v] == that_adj[u][v]
+
+    assert this_linear == that_linear, "linear terms don't match"
+    assert this_quadratic == that_quadratic, "quadratic terms don't match"
+    assert this_offset == that_offset, "offsets don't match"
