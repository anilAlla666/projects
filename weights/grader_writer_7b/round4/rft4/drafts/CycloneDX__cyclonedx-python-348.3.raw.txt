+
+        parser = PipEnvFileParser(pipenv_lock_filename=tests_pipfile_lock)
+        self.assertEqual(2, parser.component_count())
+
+        c_botocore = next(filter(lambda c: c.name == 'botocore', parser.get_components()), None)
+        self.assertIsNotNone(c_botocore)
+        c_boto3 = next(filter(lambda c: c.name == 'boto3', parser.get_components()), None)
+        self.assertIsNotNone(c_boto3)
+
+        self.assertEqual('botocore', c_botocore.name)
+        self.assertEqual('1.23.54', c_botocore.version)
+        self.assertEqual(2, len(c_botocore.external_references), f'{c_botocore.external_references}')
+        self.assertEqual(1, len(c_botocore.external_references.pop().hashes))
+
+        self.assertEqual('boto3', c_boto3.name)
+        self.assertEqual('1.20.54', c_boto3.version)
+        self.assertEqual(2, len(c_boto3.external_references), f'{c_boto3.external_references}')
