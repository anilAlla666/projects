+"""Unit tests for datamodels.
+
+Note:
+    The following code is supposed in the examples below::
+
+        from dataclasses import dataclass
+        from typing import Tuple
+        from typing_extensions import Literal
+        from xarray_dataclasses.datamodel import DataModel
+        from xarray_dataclasses.typing import Attr, Coord, Coordof, Data, Dataof
+
+"""
+
+
+from dataclasses import dataclass
+from typing import Tuple
+from typing_extensions import Literal
+from xarray_dataclasses.datamodel import DataModel
+from xarray_dataclasses.typing import Attr, Coord, Coordof, Data, Dataof
+
+
+X = Literal["x"]
+
+
+Y = Literal["y"]
+
+
+@dataclass
+class XAxis:
+    data: Data[X, int]
+    units: Attr[str] = "pixel"
+
+
+@dataclass
+class YAxis:
+    data: Data[Y, int]
+    units: Attr[str] = "pixel"
+
+
+@dataclass
+class Image:
+    data: Data[Tuple[X, Y], float]
+    mask: Coord[Tuple[X, Y], bool] = False
+    x: Coordof[XAxis] = 0
+    y: Coordof[YAxis] = 0
+
+
+@dataclass
+class Color:
+    red: Dataof[Image]
+    green: Dataof[Image]
+    blue: Dataof[Image]
+
+
+image_model = DataModel(Image)
+
+
+color_model = DataModel(Color)
+
+
+def test_image_coord() -> None:
+    mask = image_model.coords[0]
+    assert mask.name == "mask"
+    assert mask.tag == "coord"
+    assert mask.dims == ("x", "y")
+    assert mask.dtype == "bool"
+    assert mask.base is None
+    assert mask.cast == True
