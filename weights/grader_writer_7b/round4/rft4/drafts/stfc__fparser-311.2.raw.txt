+import pytest
+from fparser import api

+@pytest.mark.parametrize("source", [
+    "type real :: x",
+    "INTEGER :: y",
+    "real :: z",
+])
+def test_type_declaration(source):
+    ast = api.get_ast(source)
+    assert ast.content[0].__class__.__name__ == 'Type_Declaration_Stmt'
+
+
+@pytest.mark.parametrize("source", [
+    "character(len=10) :: a,b,c,d",
+    "CHARACTER(len=10) :: e,f,g,h",
+    "real(kind=2) :: i,j,k,l",
+])
+def test_multiple_entities_in_single_type_declaration(source):
+    ast = api.get_ast(source)
+    assert len(ast.content) == 1
+    decl_stmt = ast.content[0]
+    assert decl_stmt.__class__.__name__ == 'Type_Declaration_Stmt'
+    assert decl_stmt.entity_decl_list.__class__.__name__ == 'Entity_Decl_List'
+    assert len(decl_stmt.entity_decl_list.content) == 4
+
+
+@pytest.mark.parametrize("source", [
+    "type real :: x,y,z,w",

+])
+def test_multiple_type_declarations(source):
