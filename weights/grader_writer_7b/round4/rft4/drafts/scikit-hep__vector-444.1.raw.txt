+from __future__ import annotations
+import numpy
+import pytest
+import vector
+import vector.backends.numpy
+
+
+def test_power_of_complex_number():
+    obj = vector.obj(x=1, y=1j)
+    result = obj**2
+
+    # Assert that the result is complex and equal to the mathematical expectation
+    assert isinstance(result, vector.backends.numpy.VectorNumpy2D)
+    assert result.x == pytest.approx(-1)
+    assert result.y == pytest.approx(0)
+
+    # Test with negative exponent
+    result = obj**-1
+    assert result.x == pytest.approx(1)
+    assert result.y == pytest.approx(0)
+
+    # Test with non-complex object
+    obj = vector.obj(x=1, y=1)
+    result = obj**2
+    assert result.x == pytest.approx(1)
+    assert result.y == pytest.approx(1)
+
+    # Test with negative exponent
+    result = obj**-1
+    assert result.x == pytest.approx(1)
+    assert result.y == pytest.approx(1)
+
+    # Test with zero exponent
+    result = obj**0
+    assert result.x == pytest.approx(1)
+    assert result.y == pytest.approx(1)
