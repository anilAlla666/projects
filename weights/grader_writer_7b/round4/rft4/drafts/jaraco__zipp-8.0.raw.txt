+def add_dirs(zipfile):
+    """
+    Given a writable zipfile, inject directory entries for
+    any directories implied by the presence of children.
+    """
+    names = zipfile.namelist()
+    add = []
+    for name in names:
+        dirname = posixpath.dirname(name)
+        if dirname and dirname not in names:
+            add.append(dirname + "/")
+    for name in add:
+        zipfile.writestr(name, "", zipfile.ZIP_STORED)

+@contextlib.contextmanager
+def create_test_zipfile(filename):
+    """Create a zip file with two directories and some content."""
+    with ExitStack() as stack:
+        tmpdir = stack.enter_context(tempfile.TemporaryDirectory())
+        abcde = stack.enter_context(open(tmpdir + '/abcde', 'w'))
+        abcd = stack.enter_context(open(tmpdir + '/abcd', 'w'))
+        abcd.write('abcd')
+        abcde.write('abcdefg')
+        abcde.close()
+        abcd.close()
+        zipfile = stack.enter_context(zipp.ZipFile(filename, 'w'))
+        add_dirs(zipfile)
+        yield zipfile
+
+
+@contextlib.contextmanager
+def create_test_zipfiles(*filenames):
+    with ExitStack() as stack:
+        zips = [stack.enter_context(create_test_zipfile(f)) for f in filenames]
+        yield zips
