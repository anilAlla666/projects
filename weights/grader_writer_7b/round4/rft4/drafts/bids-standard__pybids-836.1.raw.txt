+def test_Assign():
+    var = DenseRunVariable(
+        name="rot_x",
+        values=np.arange(5., 20.),
+        run_info=RunInfo({}, 15, 1, "none", 15),
+        source='regressors',
+        sampling_rate=1
+    )
+    coll = BIDSRunVariableCollection([var], sampling_rate=1)
+
+    # Assign a single variable
+    transform.Assign(coll, "rot_x", output="d_rot_x")
+    assert "d_rot_x" in coll.variables.keys()
+
+    # Assign multiple variables
+    transform.Assign(coll, ["rot_x", "rot_y"], output=["a", "b"])
+    assert "a" in coll.variables.keys()
+    assert "b" in coll.variables.keys()
+
+    # Assign multiple variables using a single function
+    transform.Assign(coll, ["rot_x", "rot_y"], output=["c", "d"],
+                     func=lambda x, y: x + y)
+    assert "c" in coll.variables.keys()
+    assert "d" in coll.variables.keys()
+
+    # Assign multiple variables using multiple functions
+    transform.Assign(coll, ["rot_x", "rot_y"], output=["e", "f"],
+                     func=[lambda x, y: x + y, lambda x, y: x * y])
+    assert "e" in coll.variables.keys()
+    assert "f" in coll.variables.keys()
+
+    # Check that the outputs are correct
+    assert np.allclose(coll["d_rot_x"].values.values,
+                       np.arange(5., 20.))
+    assert np.allclose(coll["a"].values.values,
+                       np.arange(5., 20.))
+    assert np.allclose(coll["b"].values.values,
+                       np.arange(5., 20.))
+    assert np.allclose(coll["c"].values.values,
+                       np.arange(10., 40.))
+    assert np.allclose(coll["d"].values.values,
+                       np.arange(25., 200.))
+    assert np.allclose(coll["e"].values.values,
+                       np.arange(10., 40.))
+    assert np.allclose(coll["f"].values.values,
+                       np.arange(25., 200.))

+    # Make sure we have the right shape

+
+    # And the right values
+    assert np.allclose(merged_var.values.values,
+                       np.vstack((np.arange(5., 20.), np.arange(5., 20.))).T)
