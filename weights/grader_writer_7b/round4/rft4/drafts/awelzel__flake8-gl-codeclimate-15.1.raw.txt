+                "fixtures", "test.py")]

+\t\t\t"unused-code"
+\t\t],
+\t\t"location": {
+\t\t\t"path": "fixtures/test.py",
+\t\t\t"lines": {
+\t\t\t\t"begin": 1,
+\t\t\t\t"end": 1
