+"""
+Test the base Ellipsoid class.
+"""

+
+import warnings
+import pytest
+import numpy as np
+import numpy.testing as npt
+from .. import Ellipsoid, ELLIPSOIDS

+
+
+ELLIPSOID_NAMES = [e.name for e in ELLIPSOIDS]
+
+
+@pytest.mark.parametrize("ellipsoid", ELLIPSOIDS, ids=ELLIPSOID_NAMES)
+def test_normal_gravity_height_error(ellipsoid):
+    "Raise ValueError when computing normal gravity on internal point"
+    with pytest.raises(ValueError):
+        ellipsoid.normal_gravity(90, -1, si_units=True)
+

+def test_normal_gravity_height_warning(ellipsoid):

+    with pytest.warns(UserWarning):

+
+
+@pytest.mark.parametrize("ellipsoid", ELLIPSOIDS, ids=ELLIPSOID_NAMES)
+def test_normal_gravity_height_positive(ellipsoid):
+    "Normal gravity should be negative on internal point"
+    npt.assert_array_less(
+        0, ellipsoid.normal_gravity(90, 1, si_units=True), err_msg=str(ellipsoid)

+    )
