+import pytest
+import collections
+from pyrsistent import freeze, pmap, pset, pvector
+
+
+def test_freeze_defaultdict():
+    mydict = collections.defaultdict(list)
+    mydict["a"].append("x")
+    mydict["a"].append("y")
+    mydict["b"].append("z")
+
+    frozen = freeze(mydict)
+
+    assert type(frozen) is pmap
+    assert frozen["a"] == ["x", "y"]
+    assert frozen["b"] == ["z"]
+
+
+def test_freeze_collections_types():
+    mylist = collections.deque()
+    mylist.append("a")
+    mylist.append("b")
+    myset = collections.OrderedDict()
+    myset["c"] = 1
+    myset["d"] = 2
+
+    frozen_list = freeze(mylist)
+    frozen_set = freeze(myset)
+
+    assert type(frozen_list) is pvector
+    assert type(frozen_set) is pmap
+    assert frozen_list == pvector(["a", "b"])
+    assert frozen_set == pmap({"c": 1, "d": 2})
+
+
+if __name__ == "__main__":
+    pytest.main([__file__, "-v"])
