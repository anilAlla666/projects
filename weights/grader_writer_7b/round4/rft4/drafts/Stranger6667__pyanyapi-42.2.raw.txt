+def test_yaml_parser_vulnerability():
+    # https://github.com/Stranger6667/pyanyapi/issues/23
+    data = {'test': 'container > test'}
+    parser = YAMLInterface(data)

+        '!py !!python/object/apply:os.system ["echo hello"]'

+    with pytest.raises(InterfaceException):
+        parser.execute_method('!py !!python/object/apply:os.system ["echo hello"]')
