+class TestTelegram__remove_handler:
+    def test_remove_handler_should_not_raise_on_non_registered_handler(self, telegram):
+        telegram._handlers[MESSAGE_HANDLER_TYPE] = []
+        telegram.remove_handler(handler_type=MESSAGE_HANDLER_TYPE, func=lambda x: x)
+
+    def test_remove_handler_should_remove_handler(self, telegram):
+        telegram._handlers[MESSAGE_HANDLER_TYPE] = [lambda x: x]
+
+        telegram.remove_handler(handler_type=MESSAGE_HANDLER_TYPE, func=lambda x: x)
+
+        assert len(telegram._handlers[MESSAGE_HANDLER_TYPE]) == 0
+
+    def test_remove_handler_should_remove_handler_from_all_update_types(self, telegram):
+        telegram._handlers = {MESSAGE_HANDLER_TYPE: [lambda x: x]}
+
+        telegram.remove_handler(handler_type=MESSAGE_HANDLER_TYPE, func=lambda x: x)
+
+        assert len(telegram._handlers[MESSAGE_HANDLER_TYPE]) == 0
+
+    def test_remove_handler_should_only_remove_given_function(self, telegram):
