+def test_merge_config():
+    class Foo:
+        def __init__(self, foo_arg: str, bar: Bar):
+            self.foo_arg = foo_arg
+            self.bar = bar
+
+    class Bar:
+        def __init__(self, baz: Baz):
+            self.baz = baz
+
+    class Baz:
+        def __init__(self, qux: str):
+            self.qux = qux
+
+    parser = ArgumentParser()
+    parser.add_subclass_arguments(Foo, name="foo")
+    parser.add_subclass_arguments(Baz, name="baz")
+
+    args = parser.parse_args(["--foo.arg", "value", "--foo.bar.baz.qux", "qux-value"])
+    cfg = parser.merge_config(args)
+
+    assert args == cfg
+    assert cfg.foo_arg == "value"
+    assert cfg.foo.bar.baz.qux == "qux-value"
+
+    cfg_from = Namespace(foo_bar_baz={"qux": "other_qux"})
+    cfg = parser.merge_config(cfg_from, args)
+    assert cfg.foo_arg == "value"
+    assert cfg.foo.bar.baz.qux == "other_qux"
+

+def test_merge_config_with_choice_validation():
+    parser = ArgumentParser()
+    parser.add_argument("--arg", choices=("a", "b"))
+
+    args = parser.parse_args([])
+    cfg = parser.merge_config(args)
+    assert args == cfg
+    assert cfg.arg is None
+
+    args = parser.parse_args(["--arg=a"])
+    cfg = parser.merge_config(args)
+    assert args == cfg
+    assert cfg.arg == "a"
+
+    args = parser.parse_args(["--arg=b"])
+    cfg = parser.merge_config(args)
+    assert args == cfg
+    assert cfg.arg == "b"
+
+    args = parser.parse_args(["--arg=x"])
+    cfg = parser.merge_config(args)
+    assert args == cfg
+    assert cfg.arg == "x"
+
+    with pytest.raises(ArgumentError) as ctx:
+        parser.merge_config(Namespace(arg="x"), args)
+    assert 'Parser key "arg": "x" does not match one of the allowed values: "a", "b"' in str(ctx.value)
