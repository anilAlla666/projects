+def test_functions_are_not_part_of_comparison():
+    # see https://github.com/andialbrecht/sqlparse/issues/217
+    p = sqlparse.parse('foo = DATE(bar.baz)')[0].tokens[0]
+    assert isinstance(p, sql.Comparison)
+    assert p.tokens[0] == p.left
+    assert p.tokens[2] == p.right
