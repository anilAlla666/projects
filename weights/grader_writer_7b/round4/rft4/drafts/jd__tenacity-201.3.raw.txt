+def asynctest(callable_):
+    callable_ = asyncio.coroutine(callable_)
+
+    @six.wraps(callable_)
+    def wrapper(*a, **kw):
+        loop = asyncio.get_event_loop()
+        return loop.run_until_complete(callable_(*a, **kw))
+
+    return wrapper
+
+
+@retry
+@asyncio.coroutine
+def _retryable_coroutine(thing):
+    """A coroutine that can be retried."""
+    yield from asyncio.sleep(0.00001)
+    thing.do_something()
+
+
+@retry(stop=stop_after_attempt(2))
+@asyncio.coroutine
+def _retryable_coroutine_with_2_attempts(thing):
+    yield from asyncio.sleep(0.00001)
+    thing.go()
+
+
+class TestAsync(unittest.TestCase):
+    @asynctest
+    def test_retry(self):
+        assert asyncio.iscoroutinefunction(_retryable_coroutine)
+        thing = NoIOErrorAfterCount(5)
+        yield from _retryable_coroutine(thing)
+        assert thing.counter == thing.count
+
+    @asynctest
+    def test_stop_after_attempt(self):
+        assert asyncio.iscoroutinefunction(
+            _retryable_coroutine_with_2_attempts)
+        thing = NoIOErrorAfterCount(2)
+        try:
+            yield from _retryable_coroutine_with_2_attempts(thing)
+        except RetryError:
+            assert thing.counter == 2
+
+    def test_repr(self):
+        repr(tasyncio.AsyncRetrying())
+
+    @asynctest
+    def test_attempt_number_is_correct_for_interleaved_coroutines(self):
+
+        attempts = []
+
+        def after(retry_state):
+            attempts.append((retry_state.args[0], retry_state.attempt_number))
+
+        thing1 = NoIOErrorAfterCount(3)
+        thing2 = NoIOErrorAfterCount(3)
+        future1 = asyncio.ensure_future(
+            _retryable_coroutine.retry_with(after=after)(thing1))
+        future2 = asyncio.ensure_future(
+            _retryable_coroutine.retry_with(after=after)(thing2))
+        yield from asyncio.gather(future1, future2)
+
+        # There's no waiting on retry, only a wait in the coroutine, so the
+        # executions should be interleaved.
+        thing1_attempts = attempts[::2]
+        things1, attempt_nos1 = zip(*thing1_attempts)
+        assert all(thing is thing1 for thing in things1)
+        assert list(attempt_nos1) == [1, 2, 3]
+
+        thing2_attempts = attempts[1::2]
+        things2, attempt_nos2 = zip(*thing2_attempts)
+        assert all(thing is thing2 for thing in things2)
+        assert list(attempt_nos2) == [1, 2, 3]
+
+    @asynctest
+    def test_non_coroutine_func(self):
+        """
+        Ensure we don't get false negatives when calling
+        asyncio.iscoroutinefunction() on a wrapped coroutine func
+        """
+        @asyncio.coroutine
+        def foo():
+            pass
+
+        @retry
+        def bar():
+            pass
+
+        assert asyncio.iscoroutinefunction(foo)
+        assert not asyncio.iscoroutinefunction(bar)
+

+class TestContextManager(unittest.TestCase):
+    def test_context_manager_no_retries(self):
+        retries = Retrying(stop=tenacity.stop_after_attempt(0))
+        with retries:
+            return "success"
+
+    def test_context_manager_raises_exception(self):
+        retries = Retrying(stop=tenacity.stop_after_attempt(0))
+        with self.assertRaises(RetryError):
+            with retries:
+                raise ValueError("expected")
+
+    def test_context_manager_catches_exception(self):
+        retries = Retrying(stop=tenacity.stop_after_attempt(1))
+        with retries as retryer:
+            with self.assertRaises(ValueError):
+                retryer.call(lambda: 1 / 0)
+            self.assertEqual(retryer.statistics["attempts"], 1)
+
+    def test_context_manager_passes_arguments_to_call(self):
+        retries = Retrying(stop=tenacity.stop_after_attempt(1))
+        with retries as retryer:
+            result = retryer.call(lambda x, y: x + y, 1, 2)
+            self.assertEqual(result, 3)
+            self.assertEqual(retryer.statistics["attempts"], 1)
+
+    def test_context_manager_returns_last_value_on_success(self):
+        retries = Retrying(stop=tenacity.stop_after_attempt(1))
+        with retries as retryer:
+            result = retryer.call(lambda x: x, 42)
+            self.assertEqual(result, 42)
+            self.assertEqual(retryer.statistics["attempts"], 1)
+
+    def test_context_manager_raises_last_exception_on_failure(self):
+        retries = Retrying(stop=tenacity.stop_after_attempt(1))
+        with self.assertRaises(MyException) as cm:
+            with retries as retryer:
+                retryer.call(lambda: 1 / 0)
+        self.assertEqual(cm.exception.message, "failed")
+
+    def test_context_manager_returns_last_result_on_failure(self):
+        retries = Retrying(stop=tenacity.stop_after_attempt(1))
+        with retries as retryer:
+            result = retryer.call(lambda x: 42 / x, 0, default=42)
+            self.assertEqual(result, 42)
+            self.assertEqual(retryer.statistics["attempts"], 1)
+
+    def test_context_manager_can_be_nested(self):
+        outer_retries = Retrying(stop=tenacity.stop_after_attempt(1))
+        inner_retries = Retrying(stop=tenacity.stop_after_attempt(1))
+
+        with outer_retries as outer:
+            with inner_retries as inner:

+                inner.call(lambda: 1 / 0)

+    def test_context_manager_logs_errors(self):
+        retries = Retrying(stop=tenacity.stop_after_attempt(1))
+        with self.assertLogs(tenacity.__name__, level="ERROR") as log_cm:
