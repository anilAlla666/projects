+import pytest
+from svgpathtools import *
+
+
+def test_add_path_to_empty_group():
+    doc = Document()
+    empty = doc.add_group(group_attribs={"id": "empty"})
+    doc.add_path("M 0,0 L 1,1", group=empty)
+    assert len(empty) == 1
+
+
+def test_add_path_to_existing_group():
+    doc = Document()
+    empty = doc.add_group(group_attribs={"id": "empty"})
+    doc.add_path("M 0,0 L 1,1", group=empty)
+    doc.add_path("M 0,0 L 1,1", group=empty)
+    assert len(empty) == 2
+
+
+def test_add_path_without_group():
+    doc = Document()
+    empty = doc.add_group(group_attribs={"id": "empty"})
+    doc.add_path("M 0,0 L 1,1")
+    assert len(doc.root) == 2
+
+
+def test_add_path_with_transform_attribute():
+    doc = Document()
+    empty = doc.add_group(group_attribs={"id": "empty"})
+    doc.add_path("M 0,0 L 1,1", group=empty, attribs={"transform": "scale(2)"})
+    assert len(empty) == 1
+    assert empty[0].attrib["transform"] == "scale(2)"
+
