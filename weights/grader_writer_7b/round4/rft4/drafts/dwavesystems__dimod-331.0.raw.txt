+def assert_bqm_equal(left, right):
+    assert left.linear == right.linear
+    assert left.quadratic == right.quadratic
+    assert left.offset == right.offset
+    assert left.vartype == right.vartype
+
+
+def assert_bqm_almost_equal(left, right, places=None, check_vartype=True):
+    assert abs(left.offset - right.offset) < 1e-5, "offsets differ"
+    assert len(left.linear) == len(right.linear), "linear terms count mismatch"
+    assert len(left.quadratic) == len(right.quadratic), "quadratic terms count mismatch"
+    assert set(left.linear.keys()) == set(right.linear.keys()), "linear term keys mismatch"
+    assert set(left.quadratic.keys()) == set(right.quadratic.keys()), "quadratic term keys mismatch"
+
+    for k, v in left.linear.items():
+        assert abs(v - right.linear[k]) < 1e-5, f"mismatched linear coefficient for {k}"
+
+    for k, v in left.quadratic.items():
+        assert abs(v - right.quadratic[k]) < 1e-5, f"mismatched quadratic coefficient for {k}"
+
+    if check_vartype:
+        assert left.vartype == right.vartype, "vartypes mismatch"
