+class ParseHostTest(base.BaseTestCase):
+    def test_parse_host_unix(self):
+        assert parse_host(None) == DEFAULT_UNIX_SOCKET
+
+    def test_parse_host_tcp(self):
+        assert parse_host('localhost:2375') == 'http://localhost:2375'
+        assert parse_host('localhost:2375/') == 'http://localhost:2375/'
+
+    def test_parse_host_tcp_tls(self):
+        assert parse_host('localhost:2376', tls=True) == \
+            'https://localhost:2376'
+        assert parse_host('localhost:2376/', tls=True) == \
+            'https://localhost:2376/'
+
+    def test_parse_host_tcp_ssl(self):
+        assert parse_host('localhost:2376', tls=True) == \
+            'https://localhost:2376'
+        assert parse_host('localhost:2376/', tls=True) == \
+            'https://localhost:2376/'
+
+    def test_parse_host_windows(self):
+        assert parse_host(None, is_win32=True) == DEFAULT_NPIPE
+
+    def test_parse_host_windows_default(self):
+        assert parse_host('', is_win32=True) == DEFAULT_NPIPE
+
+    def test_parse_host_npipe(self):
+        assert parse_host('npipe:////./pipe/docker_engine') == \
+            'npipe:////./pipe/docker_engine'
+        assert parse_host('npipe:////./pipe/docker_engine/') == \
+            'npipe:////./pipe/docker_engine/'
+
+    def test_parse_host_windows_named_pipe(self):
+        assert parse_host(r'npipe:\\.\pipe\docker_engine') == \
+            'npipe:////./pipe/docker_engine'
+
+    def test_parse_host_windows_unc(self):
+        assert parse_host(r'\\.\pipe\docker_engine') == \
+            'npipe:////./pipe/docker_engine'
+
+    def test_parse_host_http(self):
