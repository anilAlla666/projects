+import unittest

+class TestAmr(unittest.TestCase):

+    def test_parse_AMR_line(self):
+        """
+        Test that parse_AMR_line parses correctly.
+        """
+        amr_text = "(w / what :ARG0-of (e / event))"
+        expected = AMR(
+            ["w", "e"],
+            ["what", "event"],
+            [[["ARG0-of", "e"]]],
+            [[["TOP", "what"], ["ARG0-of", "e"]]],
+        )
+        self.assertEqual(expected, AMR.parse_AMR_line(amr_text))
+

+        Test that compare() works correctly.

+        expected = 1.0
+        self.assertEqual(expected, AMR.compare(AMR.parse_AMR_line(amr_text1), AMR.parse_AMR_line(amr_text2)))
+
