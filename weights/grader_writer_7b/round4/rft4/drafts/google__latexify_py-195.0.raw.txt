+from __future__ import annotations
+import ast

+from latexify.codegen import expression_codegen

+@pytest.mark.parametrize(
+    "code,latex",
+    [
+        ("np.exp(a/b)", r"\exp \mathopen{}\left( \frac{a}{b} \mathclose{}\right)"),
+        (
+            "np.exp(np.sqrt(a/b))",
+            r"\exp \mathopen{}\left( \sqrt{\frac{a}{b}} \mathclose{}\right)",
+        ),
+        (
+            "np.log(a/b)",
+            r"\log \mathopen{}\left( \frac{a}{b} \mathclose{}\right)",
+        ),
+        (
+            "np.log(np.sqrt(a/b))",
+            r"\log \mathopen{}\left( \sqrt{\frac{a}{b}} \mathclose{}\right)",
+        ),
+        (
+            "np.log1p(np.exp(c/d))",
+            r"\log1p \mathopen{}\left( \exp \mathopen{}\left( \frac{c}{d} \mathclose{}\right) \mathclose{}\right)",
+        ),
+        (
+            "np.log1p(np.exp(np.sqrt(c/d)))",
+            r"\log1p \mathopen{}\left( \exp \mathopen{}\left( \sqrt{\frac{c}{d}} \mathclose{}\right) \mathclose{}\right)",
+        ),
+        (
+            "np.arctan2(np.exp(c/d), np.exp(e/f))",
+            r"\arctan2 \mathopen{}\left( \exp \mathopen{}\left( \frac{c}{d} \mathclose{}\right),"
+            r" \exp \mathopen{}\left( \frac{e}{f} \mathclose{}\right) \mathclose{}\right)",
+        ),
+        (
+            "np.arctan2(np.exp(c/d), np.exp(np.sqrt(e/f)))",
+            r"\arctan2 \mathopen{}\left( \exp \mathopen{}\left( \frac{c}{d} \mathclose{}\right),"
+            r" \exp \mathopen{}\left( \sqrt{\frac{e}{f}} \mathclose{}\right) \mathclose{}\right)",
+        ),
+    ],
+)
