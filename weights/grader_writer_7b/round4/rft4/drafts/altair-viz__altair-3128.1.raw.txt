+import pytest
+import warnings
+import json
+import numpy as np
+import pandas as pd
+from altair.utils import infer_vegalite_type, sanitize_dataframe

+
+@pytest.mark.skipif(pd.__version__ < "2.0", reason="Requires pandas v2.0+")
+def test_sanitize_pyarrow_timestamp_ms():
+    df = pd.DataFrame({"ts": pd.Timestamp("2022-01-01").to_pyarrow()})
+    df_clean = sanitize_dataframe(df)
+    assert {col.dtype.name for _, col in df_clean.items()} == {"object"}
+
+    result_python = {col_name: list(col) for col_name, col in df_clean.items()}
+    assert result_python == {
+        "ts": [
+            "2022-01-01T00:00:00+00:00",
+        ]
+    }
+
