+def test_append_columns():

+    print(tbl)
+
+    expected = (
+        "+----+----+----+\n"
+        "| One| Two| Three|\n"
+        "+====+====+======+\n"
+        "| aaa| ddd| ggg|\n"
+        "+----+----+------+\n"
+        "| bbb| eee| hhh|\n"
+        "+----+----+------+\n"
+        "| ccc| fff| iii|\n"
+        "+----+----+------+"
+    )
