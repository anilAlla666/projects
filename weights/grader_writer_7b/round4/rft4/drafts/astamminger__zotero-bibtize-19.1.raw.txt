+def test_field_parsing_with_newlines():
+    """Test parsing of BibTeX entries with field values containing commas.
+
+    This is a regression test for issue #20.
+    """
+    entry_text = r'@article{foo, title={Foo}, author={John Doe}, year={1999}, journal={Journal of Foo}, volume={10}, pages={1-2, 5}}'
+    expected_result = {'title': 'Foo', 'author': 'John Doe', 'year': '1999', 'journal': 'Journal of Foo', 'volume': '10', 'pages': '1-2, 5'}
+    result = BibEntry(entry_text).fields
+    assert(result == expected_result)
+
