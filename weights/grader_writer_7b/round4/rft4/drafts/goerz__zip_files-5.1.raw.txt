+

+import io
+import stat
+from pathlib import Path
+from zipfile import ZipFile
+from click.testing import CliRunner
+from pkg_resources import parse_version
+from test_zip_folder import _check_exit_code, _prepare_folder_with_git_excludes
+from zip_files import __version__
+from zip_files.zip_files import zip_files

+ROOT = Path(__file__).parent / 'root'
+
+
+def test_zip_files_preserve_permissions(tmp_path):
+    """Test that zip-files preserves file permissions."""
+    runner = CliRunner()
+    outfile = tmp_path / 'archive.zip'
+    files = [
+        ROOT / 'folder_with_dotfiles',
+        ROOT / 'file_executable',
+    ]
+    result = runner.invoke(
+        zip_files,
+        ['--debug', '-o', str(outfile), '-x', '*.txt']
+        + [str(f) for f in files],
+    )
+    _check_exit_code(result)
+    with ZipFile(outfile) as zipfile:
+        zipfile.debug = 3
+        assert zipfile.testzip() is None
+        info = zipfile.getinfo('folder_with_dotfiles/a/.hidden')
+        assert info.external_attr & stat.FILE_ATTRIBUTE_HIDDEN != 0
+        info = zipfile.getinfo('folder_with_dotfiles/b/5.md')
+        assert info.external_attr & stat.FILE_ATTRIBUTE_NORMAL != 0
+        info = zipfile.getinfo('folder_with_dotfiles/b/.hidden')
+        assert info.external_attr & stat.FILE_ATTRIBUTE_HIDDEN != 0
+        info = zipfile.getinfo('file_executable')
+        assert info.external_attr & stat.FILE_ATTRIBUTE_ARCHIVE != 0
+        assert info.external_attr & stat.FILE_ATTRIBUTE_READONLY != 0
+        assert info.external_attr & stat.FILE_ATTRIBUTE_SYSTEM != 0
+        assert info.external_attr & stat.FILE_ATTRIBUTE_TEMPORARY != 0
