+    def test_register_template_returns_offset_relative_to_center_pixel(self):
+        data = numpy.random.RandomState(42).randn(100, 100)
+        image_xdata = DataAndMetadata.new_data_and_metadata(data)
+        template_xdata = DataAndMetadata.new_data_and_metadata(data[40:60, 15:20])
+        ccoeff, max_pos = Core.function_register_template(image_xdata, template_xdata)
+        self.assertEqual(len(max_pos), 2)
+        self.assertTrue(numpy.allclose(max_pos, (50, 17), atol=0.1))
+        self.assertAlmostEqual(ccoeff, 1.0, places=1)
+        # verify that the offsets are relative to the center pixel
+        self.assertTrue(numpy.allclose(max_pos, (50, 17), atol=0.1))
+        self.assertEqual(max_pos[0], 50)
+        self.assertEqual(max_pos[1], 17)
+        # check that it's relative to the center
+        image_size = numpy.array([100, 100])
+        center_pixel = image_size / 2
+        expected_position = center_pixel - numpy.array(max_pos)
+        actual_position = image_xdata.get_transformed_center() - template_xdata.get_transformed_center()
+        self.assertTrue(numpy.allclose(expected_position, actual_position))
+
