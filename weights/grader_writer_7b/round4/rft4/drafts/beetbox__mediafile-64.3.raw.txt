+class TestNonexistentFields(unittest.TestCase):
+    """Test behavior of fields that don't exist in the underlying
+    Mutagen tags.
+    """
+
+    def setUp(self):
+        self.src = os.path.join(_common.RSRC, b'empty.flac')
+        self.path = os.path.join(_common.TEMP_DIR, b'empty.flac')
+        shutil.copy(self.src, self.path)
+
+    def test_nonexistent(self):
+        # A nonexistent text field should return None
+        mf = mediafile.MediaFile(self.path)
+        self.assertIsNone(mf.unknown)
+        self.assertEqual(type(mf.unknown), type(None))
+
+    def test_nonexistent_list(self):
+        # A nonexistent list field should return an empty list
+        mf = mediafile.MediaFile(self.path)
+        self.assertEqual([], mf.unknownlist)
+        self.assertEqual(type(mf.unknownlist), list)
+
+    def tearDown(self):
+        os.unlink(self.path)
