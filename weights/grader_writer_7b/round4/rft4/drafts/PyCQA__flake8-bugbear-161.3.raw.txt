+def _make_checker(file_contents, options=None):
+    """Make a checker with default options."""
+    filename = "<string>"
+    if options is not None:
+        filename = str(Path(__file__).absolute().parent / "dummy.py")
+    return BugBearChecker(
+        contents=file_contents, filename=filename, options=options
+    )
+
+
+def _check(contents, errors):
+    checker = _make_checker(contents)
+    result = list(checker.run())
+    assert len(result) == len(errors)
+    for err, expected_err in zip(result, errors):
+        assert err.message() == expected_err.message(), f"{err} != {expected_err}"
+        assert err.args == expected_err.args, f"{err.args} != {expected_err.args}"
+
+
+def _check_no_errors(contents):
+    checker = _make_checker(contents)
+    result = list(checker.run())
+    assert len(result) == 0
