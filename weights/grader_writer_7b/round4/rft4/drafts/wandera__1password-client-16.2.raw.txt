+import pytest
+from onepassword.utils import Encryption
+
+
+@pytest.mark.parametrize(
+    "input_string",
+    [
+        "test string",
+        "",
+        "short",
+        "another short string",
+        "12345678901234567890123456789012",
+    ],
+)
+def test_encryptor_padding(input_string):
+    """
+    Test the padding of input strings using the Encryption class.
+
+    :param input_string: The input string to pad.
+    :type input_string: str
+    """
+    # Create an instance of the Encryption class
+    encryption = Encryption(secret_key="a" * 32)
+
+    # Call the encode method to pad the input string
+    padded_string = encryption.encode(input_string)
+
+    # Calculate the expected length of the padded string
+    expected_length = len(input_string) + (32 - len(input_string)) % 32
+
+    # Assert that the length of the padded string matches the expected length
+    assert len(padded_string) == expected_length
