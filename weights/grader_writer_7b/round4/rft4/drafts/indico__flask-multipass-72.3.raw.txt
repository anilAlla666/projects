+def test_resolve_provider_type():
+    class Foo(AuthProvider):
+        type = 'foo'
+
+    class Bar(Foo):
+        type = 'bar'
+
+    class Moo(IdentityProvider):
+        type = 'moo'
+
+    class Quux(Moo):
+        type = 'quux'
+
+    class Other(object):
+        type = 'other'
+
+    class OtherFoo(AuthProvider):
+        type = 'otherfoo'
+
+    class OtherMoo(IdentityProvider):
+        type = 'othermoo'
+
+    assert resolve_provider_type(AuthProvider, 'foo') is Foo
+    assert resolve_provider_type(AuthProvider, 'bar') is Bar
+    assert resolve_provider_type(IdentityProvider, 'moo') is Moo
+    assert resolve_provider_type(IdentityProvider, 'quux') is Quux
+
+    assert resolve_provider_type(AuthProvider, 'foo', {'foo': Foo}) is Foo
+    assert resolve_provider_type(AuthProvider, 'bar', {'foo': Foo}) is Bar
+    assert resolve_provider_type(IdentityProvider, 'moo', {'moo': Moo}) is Moo
+    assert resolve_provider_type(IdentityProvider, 'quux', {'moo': Moo}) is Quux
+
+    with pytest.raises(KeyError):
+        resolve_provider_type(AuthProvider, 'foo', {'foo': Foo, 'moo': Moo})
+
+    with pytest.raises(KeyError):
+        resolve_provider_type(IdentityProvider, 'foo', {'foo': Foo, 'moo': Moo})
+
+    with pytest.raises(KeyError):
+        resolve_provider_type(AuthProvider, 'bar', {'moo': Moo})
+
+    with pytest.raises(KeyError):
+        resolve_provider_type(IdentityProvider, 'moo', {'foo': Foo})
+
+    with pytest.raises(KeyError):
+        resolve_provider_type(AuthProvider, 'bar', {})
+
+    with pytest.raises(KeyError):
+        resolve_provider_type(IdentityProvider, 'foo', {})
+
+    with pytest.raises(KeyError):
+        resolve_provider_type(AuthProvider, 'other', {'foo': Foo})
+
+    with pytest.raises(KeyError):
+        resolve_provider_type(IdentityProvider, 'other', {'foo': Foo})
+
+    with pytest.raises(ValueError):
+        resolve_provider_type(AuthProvider, 'otherfoo', {'foo': Foo})
+
+    with pytest.raises(ValueError):
+        resolve_provider_type(IdentityProvider, 'othermoo', {'foo': Foo})
