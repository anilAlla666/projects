+import os
+from dvc.cache import Cache
+from dvc.repo import Repo
+from dvc.system import System
+from dvc.compat import fspath
+
+
+def test_stages(tmp_dir, dvc):
+    def stages():
+        return set(stage.relpath for stage in Repo(fspath(tmp_dir)).stages)
+
+    tmp_dir.dvc_gen({"file": "a", "dir/file": "b", "dir/subdir/file": "c"})
+
+    assert stages() == {
+        "file.dvc",
+        os.path.join("dir", "file.dvc"),
+        os.path.join("dir", "subdir", "file.dvc"),
+    }
+
+    tmp_dir.gen(".dvcignore", "dir")
+
+    assert stages() == {"file.dvc"}
+
+    tmp_dir.dvc_gen({"metrics.json": "a"}, commit="init")
+    assert stages() == {"metrics.json.dvc", "file.dvc"}
+
+    tmp_dir.dvc_gen({"params.yaml": "a"}, commit="init")
+    assert stages() == {
+        "metrics.json.dvc",
+        "params.yaml.dvc",
+        "file.dvc",
+    }
+
+    tmp_dir.dvc_gen({"code.py": "a"}, commit="init")
+    assert stages() == {
+        "metrics.json.dvc",
+        "params.yaml.dvc",
+        "code.py.dvc",
+        "file.dvc",
+    }
