+def free_port():
+    with socket() as sock:
+        sock.bind(('localhost', 0))
+        return sock.getsockname()[1]

+def wait_for_server(url):
+    timeout = 10
+    end_time = monotonic() + timeout
+    while True:
+        try:
+            urlopen(url)
+            break
+        except URLError:
+            if monotonic() > end_time:
+                raise Exception('failed to connect to %s after %ss' %
+                                (url, timeout)) from None
+            sleep(0.1)
+
+
+def start_server(path, port, env):
+    cmd = [executable, '-m', 'mlrun.db.httpd', str(path)]
+    env.update({'MLRUN_httpdb__port': str(port)})
+    process = Popen(cmd, env=env, cwd=root)
+    wait_for_server(f'http://localhost:{port}')
+    url = f'http://localhost:{port}'
+    conn = HTTPRunDB(url=url, verify_ssl=False)
+    return Server(process, url, path / '.log', conn)
