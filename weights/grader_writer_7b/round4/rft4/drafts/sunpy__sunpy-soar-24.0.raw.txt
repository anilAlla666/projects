+import os
+import astropy.units as u
+import pytest
+import sunpy.map

+from sunpy.net import attrs as a

+def test_fetch_path_interpolation():
+    instrument = 'EUI'
+    time = '2021-02-11'
+    level = 1
+    product = 'EUI-FSI174-IMAGE'
+    query = a.Instrument(instrument) & a.Time(time) & a.Level(level) & a.soar.Product(product)
+    res = Fido.search(query)
+    files = Fido.fetch(res, path=f'data/{instrument}/')
+    assert len(files) == 1
+    fname = files[0]
+    assert os.path.basename(fname).startswith('SOA210211T')
+    assert u.allclose(os.path.getsize(fname) * u.byte,
+                      res[0, 0]['Filesize'], atol=1e-3*u.Mbyte)
+    # Smoke test that we can read this into a map
+    sunpy.map.Map(fname)
+
