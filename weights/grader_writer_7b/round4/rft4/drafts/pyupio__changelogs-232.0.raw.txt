+from __future__ import unicode_literals
+from changelogs.finder import contains_project_name
+from urllib.parse import urlparse
+
+
+def test_validate_url():
+    assert urlparse("http://example.com").netloc == "example.com"
+    assert urlparse("https://example.com").netloc == "example.com"
+    assert urlparse("git@github.com:foobar/example.git").netloc == "github.com"
+    assert urlparse("http://example.com").path == "/"
+    assert urlparse("/").path == "/"
+    assert urlparse("").path == ""
+    assert urlparse("ftp://example.com").path == "/"
+
+
+def test_find_repo_urls():
+    call = changelogs.find_repo_urls
+    assert list(call(None, "foo", [])) == []
+    assert len(list(call(None, "foo", ["http://example.com"])) > 0
+    assert len(list(call(None, "foo", ["http://example.com/foo"])) > 0
+    assert len(list(call(None, "foo", ["http://example.com/foo/bar"])) > 0
+    assert len(list(call(None, "foo", ["http://example.com/foo/bar/baz"])) > 0

+    assert len(list(call(None, "foo", ["http://example.com/foo/bar/baz/quux/txt"])) > 0
