+class TestUrltools(unittest.TestCase):
+    def test_unquote(self):
+        self.assertEqual(unquote('foo%23bar'), 'foo#bar')
+        self.assertEqual(unquote('foo%23bar', ['#']), 'foo%23bar')
+        with self.assertRaises(TypeError):
