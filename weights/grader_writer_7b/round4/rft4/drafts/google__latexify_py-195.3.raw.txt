+        ("exp((x/y)**z)", r"\exp \mathopen{}\left( \frac{x}{y} \mathclose{}\right)^z"),
+        ("sin((x/y)**z)", r"\sin \mathopen{}\left( \frac{x}{y} \mathclose{}\right)^z"),

+    ],
+)
