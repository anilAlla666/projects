+def loadTests():
+    '''

+    '''
+    suite = unittest.TestSuite()
+    for test in i386Tests:
+        suite.addTest(TestI386(test))

+class TestI386(unittest.TestCase):
+    '''
+    Test a single i386 opcode
+    '''
+    def __init__(self, testdict):
+        super().__init__()
+        self.testdict = testdict
+
+    def setUp(self):
+        self.emu = envi.getArchModule('i386').getEmulator()
+        self.emu.setProgramCounter(0x4000)
+        self.emu.setStackCounter(STACKBASE)
+        self.emu.setSegmentRegisterByName('ds', 0x0)
+        self.emu.setSegmentRegisterByName('ss', 0x0)
+        self.emu.setSegmentRegisterByName('cs', 0x0)
+        self.emu.setSegmentRegisterByName('es', 0x0)
+        self.emu.setSegmentRegisterByName('fs', 0x0)
+        self.emu.setSegmentRegisterByName('gs', 0x0)
+        self.emu.setFlagByName('eflags', 0)
+        if 'setup' in self.testdict:
+            regs, mems = self.testdict['setup']
+            for regname, regval in regs.items():
+                self.emu.setRegisterByName(regname, regval)
+            for addr, data in mems.items():
+                self.emu.writeMemory(addr, data)
+        self.emu.setProgramCounter(0x4000)
+
+    def checkResults(self):
+        for regname, expected in self.testdict['tests'].items():
+            actual = self.emu.getRegisterByName(regname)
+            self.assertEqual(actual, expected, "%s != %s" % (actual, expected))
+
+    def runTest(self):
+        self.emu.executeOpcode(self.emu.archParseOpcode(binascii.unhexlify(self.testdict['bytes']), 0, 0))
+        if 'tests' in self.testdict:
+            self.checkResults()
