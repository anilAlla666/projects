+    async def foo():
+        return "foo"
+
+    executor = ThreadPoolExecutor(max_workers=1)
+    with Injector([lambda binder: binder.async_bind(foo)]) as injector:
+        result = injector.run_async(foo)
+        assert result.result() == "foo"
+
+        # Check that we can call it multiple times without raising errors
+        for i in range(10):
+            assert result.result() == "foo"

+    # Check that it still works even after clearing
+    injector.clear()
