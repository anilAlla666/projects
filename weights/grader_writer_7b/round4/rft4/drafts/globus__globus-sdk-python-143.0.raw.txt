+        super(AuthClientTests, self).setUp()
+

+        self.access_token = token_res["access_token"]

+        self.ac = globus_sdk.AuthClient(
+            authorizer=globus_sdk.AccessTokenAuthorizer(self.access_token),
+            client_id=client_id)

+    def test_oauth2_start_flow_native_app(self):
+        """
+        Tests starting the native app flow

+                              globus_sdk.auth.GlobusNativeAppFlowManager)

+                         self.ac.oauth2_start_flow_native_app().auth_client.client_id)
