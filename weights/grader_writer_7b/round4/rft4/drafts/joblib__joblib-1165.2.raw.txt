+def test_ignore_decorator():
+    def decorator(f):
+        @functools.wraps(f)
+        def wrapper(*args, **kwargs):
+            return f(*args, **kwargs)
+        return wrapper
+
+    @memoized_func
+    @decorator
+    def func(a, b):
+        return a + b
+
+    assert func.ignore_list == ['a', 'b']
+
