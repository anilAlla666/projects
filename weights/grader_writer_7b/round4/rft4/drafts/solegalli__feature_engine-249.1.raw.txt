+import numpy as np
+import pandas as pd
+import pytest
+from feature_engine.creation import MathematicalCombination
+
+
+def test_ignore_null_values_in_combinations(df_with_nan):
+    transformer = MathematicalCombination(
+        variables_to_combine=["A", "C"],
+        ignore_na=True,
+    )
+
+    X_t = transformer.fit_transform(df_with_nan)
+
+    assert (
+        X_t.loc[
+            [2, 3, 4], ["A+C", "A-C", "A*C", "A/C", "A^C", "A%C"]
+        ].values.sum()
+        == df_with_nan.dropna(subset=["A", "C"]).loc[:, ["A", "C"]].values.sum()
+    )
+
+
