+        # Get the property
+        prop = component.test_property

+
+        # Change the property
+        ipopo.bind("requires_var_filter", {"test_property": "second"})
+
+        # Update the property
+        ipopo.set_property("component_1", "test_property", "third")

+        # Check the result
+        prop = component.test_property

+        # Update the property again
+        ipopo.set_property("component_1", "test_property", "fourth")
+
+        # Check the result

+        self.assertEqual(prop, "fourth")
+
+        # Try with a bad filter
