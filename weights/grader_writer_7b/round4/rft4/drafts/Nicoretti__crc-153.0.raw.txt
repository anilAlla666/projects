+class TestTableBasedRegister:
+    def test_init(self):
+        tbr = TableBasedRegister(Configuration())
+        assert tbr.register == 0
+
+    def test_update_with_bytes(self):
+        data = b"Hello World!"
+        tbr = TableBasedRegister(Configuration())
+        result = tbr.update(data)
+        assert result == 138
+        assert tbr.register == 138
+
+    def test_update_with_iterable(self):
+        data = iter([b"H", b"e", b"l", b"l", b"o"])
+        tbr = TableBasedRegister(Configuration())
+        result = tbr.update(data)
+        assert result == 138
+        assert tbr.register == 138
+
+    def test_update_with_int(self):
+        tbr = TableBasedRegister(Configuration())
+        result = tbr.update(138)
+        assert result == 138
+        assert tbr.register == 138
+
+    def test_reverse(self):
+        tbr = TableBasedRegister(Configuration(), 0x0102030405060708)
+        assert tbr.reverse() == 0x0807060504030201
+
+    def test_digest(self):
+        tbr = TableBasedRegister(
+            Configuration(reverse_output=True), 0x0102030405060708
