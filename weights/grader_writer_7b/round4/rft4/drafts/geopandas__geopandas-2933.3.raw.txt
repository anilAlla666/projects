+def test_merge_preserve_geometry_columns():
+    # https://github.com/geopandas/geopandas/issues/2931
+    point = shapely.Point(600000, 5000000)
+    df_1 = geopandas.GeoDataFrame(
+        data={"geom": [shapely.Point([1, 2])]},
+        geometry="geom",
+        crs="25832",
+    )
+    df_2 = geopandas.GeoDataFrame(
+        data={"geom": [point]},
+        geometry="geom",
+        crs="25832",
+    )
+    df_merged = df_1.merge(df_2, left_index=True, right_index=True)
+    assert type(df_merged) == geopandas.GeoDataFrame
+    assert df_merged._geometry_column_name == "geom"
+
+    df_merged = df_1.merge(df_2, left_index=True, right_index=True, suffixes=(None, "_r"))
+    assert type(df_merged) == geopandas.GeoDataFrame
+    assert df_merged._geometry_column_name == "geom"
+
+    df_merged = df_1.merge(df_2, how="outer", left_index=True, right_index=True)
+    assert type(df_merged) == geopandas.GeoDataFrame
+    assert df_merged._geometry_column_name == "geom"
