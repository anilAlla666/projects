+try:
+    import json
+except ImportError:
+    import simplejson as json
+
+
+import commentjson
+import os
+import unittest
+from six import iteritems
+
+
+class TestCommentJson(unittest.TestCase):
+    def setUp(self):
+        self.test_json = {}
+        self.path = os.path.dirname(os.path.abspath(__file__))
+        self.files = ('sample', 'line_comment', 'inline_last_float',
+                      'inline_last_int', 'nested_object', 'string_with_hash',
+                      'string_with_inline_comment',
+                      'inline_has_special_characters',
+                      'array_with_hash',
+                      'inline_last_quote')
+
+        for file_ in self.files:
+            fpath = os.path.join(self.path, file_)
+
+            with open('%s-uncommented.json' % fpath) as fp:
+                uncommented = fp.read()
+            with open('%s-commented.json' % fpath) as fp:
+                commented = fp.read()
+
+            self.test_json.update({
+                file_: {
+                    'uncommented': uncommented,
+                    'commented': commented,
+                },
+            })
+
+    def tearDown(self):
+        for file_ in self.files:
+            os.remove(os.path.join(self.path, '%s-commented.json' % file_))
+
+    def test_dumps_trailing_comma(self):
+        test_dict = dict(a=[1, 2], b={'c': 3})
+        j_dump = json.dumps(test_dict)
+        c_dump = commentjson.dumps(test_dict)
+
+        assert c_dump == j_dump, c_dump
+
+    def test_dumps_trailing_comma_with_sort_keys(self):
+        test_dict = dict(a=[1, 2], b={'c': 3})
+        j_dump = json.dumps(test_dict, sort_keys=True)
+        c_dump = commentjson.dumps(test_dict, sort_keys=True)
+
+        assert c_dump == j_dump, c_dump
+
+    def test_dumps_trailing_comma_with_indent(self):
+        test_dict = dict(a=[1, 2], b={'c': 3})
+        j_dump = json.dumps(test_dict, indent=4)
+        c_dump = commentjson.dumps(test_dict, indent=4)
+
+        assert c_dump == j_dump, c_dump
+
+    def test_dumps_trailing_comma_with_indent_and_sort_keys(self):
+        test_dict = dict(a=[1, 2], b={'c': 3})
+        j_dump = json.dumps(test_dict, indent=4, sort_keys=True)
+        c_dump = commentjson.dumps(test_dict, indent=4, sort_keys=True)
+
+        assert c_dump == j_dump, c_dump
+
+    def test_dumps_trailing_comma_with_indent_and_separators(self):
+        test_dict = dict(a=[1, 2], b={'c': 3})
+        j_dump = json.dumps(test_dict, indent=4, separators=(',', ': '))
+        c_dump = commentjson.dumps(test_dict, indent=4, separators=(',', ': '))
+
+        assert c_dump == j_dump, c_dump
+
+    def test_dumps_trailing_comma_with_indent_and_sort_keys_and_separators(self):
+        test_dict = dict(a=[1, 2], b={'c': 3})
+        j_dump = json.dumps(test_dict, indent=4, sort_keys=True,
+                            separators=(',', ': '))
+        c_dump = commentjson.dumps(test_dict, indent=4, sort_keys=True,
+                                   separators=(',', ': '))
+
+        assert c_dump == j_dump, c_dump
+
+    def test_load(self):
+        for file_ in self.files:
+            rfp = open(os.path.join(self.path, '%s-commented.json' % file_),
+                       'r')
+            uncommented = self.test_json[file_]['uncommented']
+            assert commentjson.load(rfp) == json.loads(uncommented)
+            rfp.close()
+
+    def test_load_trailing_comma(self):
+        rfp = open(os.path.join(self.path, 'trailing-comma-commented.json'),
+                   'r')
+        uncommented = self.test_json['sample']['uncommented']
+        assert commentjson.load(rfp) == json.loads(uncommented)
+        rfp.close()
+
+    def test_load_trailing_comma_with_kwargs(self):
+        def test_hook(loaded_dict):
+            return {}
+
+        test_kwargs = dict(object_hook=test_hook)
+        rfp = open(os.path.join(self.path, 'sample-commented.json'), 'r')
+
+        self.assertEqual(commentjson.load(rfp, **test_kwargs), {})
+        rfp.close()
+
+    def test_load_throws_exception(self):
+        wfp = open(os.path.join(self.path, 'test.json'), 'w')
+        wfp.write('Unserializable text.')
+        wfp.close()
+
+        rfp = open(os.path.join(self.path, 'test.json'), 'r')
+        self.assertRaises(commentjson.JSONLibraryException, commentjson.load,
+                          rfp)
+        rfp.close()
+
+
+if __name__ == '__main__':
+    unittest.main()
