+class TestShortcuts(TestCase):
+    def test_shortcut_exactlyone(self):
+        mgr = get_env().formula_manager
+        a = mgr.Symbol('a', BOOL)
+        b = mgr.Symbol('b', BOOL)
+        c = mgr.Symbol('c', BOOL)
+
+        result = mgr.ExactlyOne(a, b, c)
+
+        expected = mgr.And(mgr.Or(a, b, c),
+                           mgr.Not(mgr.Or(mgr.And(a, b), mgr.And(b, c), mgr.And(c, a))))
+
+        self.assertEqual(result, expected)
+
+    def test_shortcut_atmostone(self):
+        mgr = get_env().formula_manager
+        a = mgr.Symbol('a', BOOL)
+        b = mgr.Symbol('b', BOOL)
+        c = mgr.Symbol('c', BOOL)
+
+        result = mgr.AtMostOne(a, b, c)
+
+        expected = mgr.And(mgr.Not(mgr.Or(mgr.And(a, b), mgr.And(b, c), mgr.And(c, a))),
+                           mgr.Or(a, b, c))
+
+        self.assertEqual(result, expected)
+
+    def test_shortcut_all_different(self):
