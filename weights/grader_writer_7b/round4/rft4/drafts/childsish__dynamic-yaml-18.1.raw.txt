+from argparse import ArgumentParser
+from unittest import TestCase, main
+from dynamic_yaml import load
+
+
+class TestDynamicYaml(TestCase):
+    def test_list_resolution(self):
+        config = '''
+        - a
+        - b
+        - '{root[0]}'
+        '''
+        res = load(config)
+        self.assertEqual('a', res[0])
+        self.assertEqual('b', res[1])
+        self.assertEqual('a', res[2])
+
+    def test_list_methods(self):
+        config = '''
+        - a
+        - b
+        - '{root[0]}'
+        '''
+        res = load(config)
+        self.assertEqual(['a', 'b', 'a'], list(res))
+        self.assertSetEqual({'a', 'b'}, set(res))
+        self.assertEqual(['a', 'b', 'a'], res.to_list())
+        self.assertSetEqual({'a', 'b'}, res.to_set())
