+def test_multistrategy_dispatch_no_mro_fallback():
+    """
+    Test that subclasses don't cause mro issues when falling back to
+    function dispatch.
+    """
+    class Bar(Foo):
+        pass
+
+    dispatch = MultiStrategyDispatch(_fallback)
