+def test_make_local_storage_parallel(caplog):
+    "Check that we can make a cache directory safely in parallel"
+    from multiprocessing.pool import Pool
+
+    caplog.set_level(10)  # debug level
+    with TemporaryDirectory() as tmp:
+        path = os.path.join(tmp, "testcache")
+
+        def mkstorage(i):
+            make_local_storage(path)
+            return i
+
+        pool = Pool(processes=4)
+        result = pool.map(mkstorage, range(4))
+        pool.close()
+        pool.join()
+        assert len(result) == 4
+        assert all(isinstance(r, int) for r in result)
+        assert Path(path).is_dir()
+        # Check that there were no warnings
+        log_lines = [record.message for record in caplog.records]
+        assert len(log_lines) == 0
