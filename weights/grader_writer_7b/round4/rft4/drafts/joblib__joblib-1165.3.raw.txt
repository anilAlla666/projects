+def test_signature_with_decorated_function():
+    def dec(f):
+        @functools.wraps(f)
+        def wrapper(*args, **kwargs):
+            return f(*args, **kwargs)
+        return wrapper
+
+    @dec
+    def foo(a, b):
+        pass
+
+    _, sig = format_signature(foo)
+    assert 'a' in sig
+    assert 'b' in sig
