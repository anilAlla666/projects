+def local_basic_sweeper():
+    class LocalBasicSweeper(Sweeper):
+        def __init__(self, launcher):
+            self.launcher = launcher
+
+        def setup(self, config_name: str, job_results: list):
+            pass
+
+        def finalize(self, config_name: str, job_results: list):
+            pass
+
+        def run_jobs(
+            self,
+            job_batch: list,
+            arguments: list,
+            initial_job_idx: int,
+            num_total_jobs: int,
+        ):
+            parser = OverridesParser.create(config_loader=self.launcher.hydra_context.config_loader)
+            override = parser.parse_overrides(arguments)[0]
+            key = override.get_key_element()
+            sweep = [f"{key}={val}" for val in override.sweep_string_iterator()]
+            overrides = [[x] for x in sweep]
+
+            returns = []
+            for i, batch in enumerate(overrides):
+                result = self.launcher.launch([batch], initial_job_idx=i)[0]
+                returns.append(result)
+
+            return [returns]
+
+    return LocalBasicSweeper
