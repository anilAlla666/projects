+class TestOSFS(FSTestCases, unittest.TestCase):
+    """Test OSFS implementation."""
+
+    def make_fs(self):
+        temp_dir = tempfile.mkdtemp("fstestosfs")
+        return osfs.OSFS(temp_dir)
+
+    def destroy_fs(self, fs):
+        self.fs.close()
+        try:
+            shutil.rmtree(fs.getsyspath("/"))
+        except OSError:
+            # Already deleted
+            pass
+
+    def setUp(self):
+        super(TestOSFS, self).setUp()
+        self.addCleanup(self.destroy_fs, self.fs)
+
+    def test_unicode_filenames(self):
+        """Check Unicode filenames can be handled."""
+        with self.fs.open_text("\u2660\u2662\u2664\u2666.txt", "w") as fh:
+            fh.write(u"\u2660\u2662\u2664\u2666")
+        with self.fs.open_text("\u2660\u2662\u2664\u2666.txt", "r") as fh:
+            data = fh.read()
+            self.assertEqual(data, u"\u2660\u2662\u2664\u2666")
+
+    def test_am_pm_filenames(self):
+        """Check that filenames containing AM/PM are correctly parsed."""
+        with self.fs.open_binary("clock_at_AM_or_PM.txt", "w") as fh:
+            fh.write(b"foobar")
+        stat_info = self.fs.getinfo("clock_at_AM_or_PM.txt", namespaces=["basic", "details"])
+        self.assertEqual(stat_info.size, 6)
+        self.assertEqual(stat_info.name, "clock_at_AM_or_PM.txt")
+
+    def test_invalid_chars_in_filename(self):
+        """Check InvalidCharsInPath error is raised on bad encoding."""
+
+        with mock.patch("fs.osfs.fsencode") as fsencode:
+            fsencode.side_effect = lambda error: "–".encode("ascii")
+            with self.assertRaises(errors.InvalidCharsInPath):
+                with self.fs.open("13 – Marked Register.pdf", "wb") as fh:
+                    fh.write(b"foo")
+
+    def test_consume_geturl(self):
+        self.fs.create("foo")
+        try:
+            url = self.fs.geturl("foo", purpose="fs")
+        except errors.NoURL:
+            self.assertFalse(self.fs.hasurl("foo"))
+        else:
+            self.assertTrue(self.fs.hasurl("foo"))
+
+        # Should not throw an error
