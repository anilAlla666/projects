+from __future__ import annotations
+import sys
+import numpy
+import pytest
+import vector
+import vector.backends.object
+
+
+numba = pytest.importorskip("numba")
+
+
+import vector.backends._numba_object  # noqa: E402
+
+
+pytestmark = pytest.mark.numba
+
+
+def test_power_of_vector():
+    # @numba.njit
+    def get_pow_vec(vec, exponent):
+        return vec**exponent
+
+    # @numba.njit
+    def get_pow_vec_obj(vec, exponent):
+        return vec**exponent
+
+    # @numba.njit
+    def get_pow_vec_num(vec, exponent):
+        return vec**exponent
+
+    # @numba.njit
+    def get_pow_vec_obj_num(vec, exponent):
+        return vec**exponent
+
+    vec = vector.obj(x=1, y=2, z=3, t=4)
+
+    out = get_pow_vec(vec, 2)
+    assert isinstance(out, vector.backends.object.ScalarObject)
+    assert out.value == pytest.approx(29)
+
+    out = get_pow_vec_obj(vec, 2)
+    assert isinstance(out, vector.backends.object.ScalarObject)
+    assert out.value == pytest.approx(29)
+
+    out = get_pow_vec_num(vec, 2)
+    assert isinstance(out, vector.backends.object.ScalarObject)
+    assert out.value == pytest.approx(29)
+
+    out = get_pow_vec_obj_num(vec, 2)
+    assert isinstance(out, vector.backends.object.ScalarObject)
+    assert out.value == pytest.approx(29)
+
+    out = get_pow_vec(vec, 0.5)
+    assert isinstance(out, vector.backends.object.ScalarObject)
+    assert out.value == pytest.approx(2.1213203435596424)
+
+    out = get_pow_vec_obj(vec, 0.5)
+    assert isinstance(out, vector.backends.object.ScalarObject)
+    assert out.value == pytest.approx(2.1213203435596424)
+
+    out = get_pow_vec_num(vec, 0.5)
+    assert isinstance(out, vector.backends.object.ScalarObject)
+    assert out.value == pytest.approx(2.1213203435596424)
+
+    out = get_pow_vec_obj_num(vec, 0.5)
+    assert isinstance(out, vector.backends.object.ScalarObject)
+    assert out.value == pytest.approx(2.1213203435596424)
+
+    out = get_pow_vec(vec, 2j)
+    assert isinstance(out, vector.backends.object.ScalarObject)
+    assert out.value == pytest.approx(2j)
+
+    out = get_pow_vec_obj(vec, 2j)
+    assert isinstance(out, vector.backends.object.ScalarObject)
+    assert out.value == pytest.approx(2j)
