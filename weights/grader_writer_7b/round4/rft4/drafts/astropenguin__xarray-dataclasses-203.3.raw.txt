+from dataclasses import dataclass
+from typing import Tuple
+
+
+from typing_extensions import Literal
+
+
+from xarray_dataclasses.datamodel import DataModel
+from xarray_dataclasses.typing import Attr, Coord, Coordof, Data, Dataof
+
+
+X = Literal["x"]
+
+
+Y = Literal["y"]
+
+
+@dataclass
+class XAxis:
+    data: Data[X, int]
+    units: Attr[str] = "pixel"
+
+
+@dataclass
+class YAxis:
+    data: Data[Y, int]
+    units: Attr[str] = "pixel"
+
+
+@dataclass
+class Image:
+    data: Data[Tuple[X, Y], float]
+    mask: Coord[Tuple[X, Y], bool] = False
+    x: Coordof[XAxis] = 0
+    y: Coordof[YAxis] = 0
+
+
+@dataclass
+class RedChannel(Image):
+    pass
+
+
+@dataclass
+class GreenChannel(Image):
+    pass
+
+
+@dataclass
+class BlueChannel(Image):
+    pass
+
+
+image_model = DataModel[Image]()
+
+
+color_model = DataModel[
+    Tuple[RedChannel, GreenChannel, BlueChannel],
+    "red",
+    "green",
+    "blue",
+]()
+
+
+def test_image_data() -> None:
+    data = image_model.data_vars[0]
+    assert data.name == "data"
+    assert data.tag == "data"
+    assert data.dims == ("x", "y")
+    assert data.dtype == "float"
+    assert data.base is None
+    assert data.cast == True
