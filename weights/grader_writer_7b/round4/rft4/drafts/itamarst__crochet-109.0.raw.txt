+class TestThreadedEventLoopMixin:
+    @pytest.mark.skipif(sys.platform != 'win32',
+                        reason='test specific to Windows')
+    def test_log_observer_crashes_in_windows(self, loop, monkeypatch):
+        """Ensure we can handle a broken log observer"""
+        def crash(*args):
+            raise Exception("Crash!")
+        monkeypatch.setattr(
+            'logging.StreamHandler.emit', crash)
+        observer = ThreadLogObserver(crash)
+        observer(None)
+
+    def test_result_registry_has_weakrefs_to_results(self, loop):
+        result = EventualResult(loop._reactor, lambda: None)
+        registry = loop._result_registry
+        ref = weakref.ref(result)
+        del result
+        assert ref() is not None
+        registry.clear()
+        assert ref() is None
+
+    def test_threaded_event_loop_mixin_adds_threads_to_main_thread_group(self,
+                                                                        loop):
+        group = threading.current_thread().ident
+        mixin = ThreadedEventLoopMixin(group)
+        mixin.start()
+        assert threading.current_thread().ident == group
+        mixin.stop()
