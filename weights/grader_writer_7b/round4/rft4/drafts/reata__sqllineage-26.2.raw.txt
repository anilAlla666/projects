+import pytest
+from sqllineage.core import LineageParser
+
+
+@pytest.mark.parametrize(
+    "sql",
+    [
+        """
+        -- comment
+        """,
+        """
+        -- comment
+        ;;
+        """,
+        """
+        /* block comment */
+        """,
+        """
+        /* block comment */
+        ;;
+        """,
+        """
+        /* block comment */ ;
+        """,
+    ],
+)
+def test_empty_statement(sql):
+    lp = LineageParser(sql)
+    assert len(lp.statements) == 0
+    assert len(lp.source_tables) == 0
+    assert len(lp.target_tables) == 0
+    assert len(lp.tmp_tables) == 0
+    assert len(lp.stmt_to_src_tables) == 0
+    assert len(lp.stmt_to_tgt_tables) == 0
