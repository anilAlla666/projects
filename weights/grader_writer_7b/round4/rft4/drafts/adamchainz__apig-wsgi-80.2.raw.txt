+def make_event(
+    method="GET",
+    qs_params=None,
+    headers=None,
+    body="",
+    binary=False,
+    request_context=None,
+):
+    if headers is None:
+        headers = {"Host": "example.com"}
+
+    event = {
+        "httpMethod": method,
+        "path": "/",
+        "queryStringParameters": qs_params,
+        "headers": headers,
+    }
+
+    if body:
+        event["body"] = body
+        event["isBase64Encoded"] = binary
+
+    if request_context is not None:
+        event["requestContext"] = request_context
+
+    return event
+
