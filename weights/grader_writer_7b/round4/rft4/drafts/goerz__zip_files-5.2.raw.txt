+def test_zip_files_preserve_permissions(tmp_path):
+    """Test that zip-files preserves permissions."""
+    runner = CliRunner()
+    folder = tmp_path / 'archive'
+    folder.mkdir()
+    outfile = tmp_path / 'archive.zip'
+    files = [
+        folder / 'a',
+        folder / 'b',
+        folder / 'c',
+    ]
+    files[0].mkdir(mode=0o755 | 0o400)
+    files[1].touch(mode=0o755 | 0o400)
+    files[2].touch(mode=0o755 | 0o400)
+    for f in files:
+        f.chmod(0o755 | 0o400)
+    result = runner.invoke(
+        zip_files,
+        ['--debug', '-o', str(outfile)]
+        + [str(f) for f in files],
+    )
+    _check_exit_code(result)
+    expected_files = [
+        'a',
+        'b',
+        'c',
+    ]
+    with ZipFile(outfile) as zipfile:
+        zipfile.debug = 3
+        assert zipfile.testzip() is None
+        assert set(expected_files) == set(zipfile.namelist())
+        assert zipfile.infolist()[0].external_attr == 0o755 << 16
+        assert zipfile.infolist()[1].external_attr == 0o755 << 16
+        assert zipfile.infolist()[2].external_attr == 0o755 << 16
