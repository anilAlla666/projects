+def test_discard_in_transformer_non_recursive(Transformer):
+    grammar = (
+        r"""
+        start: expr NEWLINE
+            | empty
+        NEWLINE: "\n"
+        expr: term (("+" | "-") term)*
+        term: factor (("*" | "/") factor)*
+        factor: NUMBER | "(" expr ")" | "discard"
+        discard: "discard" -> discard
+        %import common.NUMBER
+        %import common.WS
+        %ignore WS
+        """
+    )
+
+    parser = Lark(grammar, parser="lalr")
+
+    tree = parser.parse("1 + 2\n")
+    transformer = Transformer()
+    transformed_tree = transformer.transform(tree)
+    assert transformed_tree.pretty() == "expr\n  term\n    factor\n      NUMBER\n        1\n  +\n  term\n    factor\n      NUMBER\n        2\nNEWLINE"
+
+    tree = parser.parse("(1 + discard)\n")
+    transformer = Transformer()
+    transformed_tree = transformer.transform(tree)

+
+    tree = parser.parse("(1 + discard)\n")
+    transformer = Transformer()
+    transformed_tree = transformer.transform(tree)

+
+    tree = parser.parse("discard\n")
