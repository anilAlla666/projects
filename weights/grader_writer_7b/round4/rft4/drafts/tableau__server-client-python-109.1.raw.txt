+class TSC:
+    class Pager(object):
+        def __init__(self, resource):
+            self._resource = resource
+            self._request_options = {}
+
+        @property
+        def request_options(self):
+            return self._request_options
+
+        def __len__(self):
+            return self._resource.request_options['pagination']['totalAvailable']
+
+        def __getitem__(self, index):
+            if index >= len(self):
+                raise IndexError("Index out of range")
+
+            return self._resource[index]
+
+        def __iter__(self):
+            for i in range(len(self)):
+                yield self[i]
+
+    class Users(TSC.Resource):
+        def __init__(self, **kwargs):
+            super().__init__(**kwargs)
+            self._api_endpoint = "/api/{}/sites/{}/users".format(
+                self._connection.server_info.api_version, self._site.id)
+            self._items = set()
+            self._pagination = {'totalAvailable': 0}
+            self._request_options = {'pagination': {}}
+
+        @property
+        def request_options(self):
+            return self._request_options
+
+        def populate_all(self):
+            while True:
+                self._request_options['pagination'].update({
+                    'pageNumber': self._pagination['pageNumber'],
+                    'pageSize': self._pagination['pageSize'],
+                })
+                self._request_options['filter'] = {}
+                items = self._connection.get(self._api_endpoint, self._request_options)
+                self._items.update(items)
+                self._pagination = items.pagination
+                if items.pagination['totalPages'] > self._pagination['pageNumber']:
+                    self._request_options['pagination']['pageNumber'] += 1
+                else:
+                    break
