+from __future__ import annotations
+import numpy
+import pytest
+import vector
+
+
+def test_pow():
+    obj = vector.obj(x=1, y=1)
+    assert obj**2 == 2
+
+    obj = vector.obj(x=1, y=-1)
+    assert obj**2 == 2
+
+    obj = vector.obj(x=-1, y=-1)
+    assert obj**2 == 2
+
+    obj = vector.obj(x=-1, y=1)
+    assert obj**2 == 2
+
+    obj = vector.obj(x=0, y=1)
+    assert obj**2 == 1
+
+
+def test_pow_backwards_compatibility():
+    obj = vector.obj(E=1, px=1, py=1, pz=1)
+    assert obj**2 == -2
+
+    obj = vector.obj(mass=1, px=1, py=1, pz=1)
+    assert obj**2 == -2
+
+    obj = vector.obj(m=1, px=1, py=1, pz=1)
+    assert obj**2 == -2
+
+    obj = vector.obj(E=1, px=1, py=1, pz=1)
+    assert obj.mass**2 == -2
+
+    obj = vector.obj(E=1, px=1, py=1, pz=1)
+    assert obj.m**2 == -2
+
+    obj = vector.obj(E=1, px=1, py=1, pz=1)
+    assert obj.mass2 == -2
+
+    obj = vector.obj(E=1, px=1, py=1, pz=1)
+    assert obj.m2 == -2
