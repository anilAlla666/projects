+import datetime
+from unittest import TestCase
+import jsons
+
+
+class TestJsonSerializable(TestCase):
+    def test_jsonserializable_with_timezone(self):
+        class Bar(jsons.JsonSerializable):
+            def __init__(self, baz: datetime.datetime, quux: datetime.timedelta):
+                self.baz = baz
+                self.quux = quux
+
+        class Foo(jsons.JsonSerializable):
+            def __init__(self, bar: Bar, ham: str):
+                self.bar = bar
+                self.ham = ham
+
+        loaded = Foo.from_json({
+            'bar': {
+                'baz': '2020-01-01T00:00:00+00:00',
+                'quux': 42,
+            },
+            'ham': 'spam',
+        })
+        self.assertEqual('spam', loaded.ham)
+        self.assertEqual('2020-01-01T00:00:00+00:00', loaded.bar.baz.isoformat())
+        self.assertEqual(datetime.timedelta(seconds=42), loaded.bar.quux)
+
