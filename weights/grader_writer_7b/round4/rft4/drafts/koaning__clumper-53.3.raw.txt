+import os
+from pathlib import Path
+from tempfile import NamedTemporaryFile
+import pytest
+from clumper import Clumper
+
+
+@pytest.mark.parametrize("extension", [".json", ".jsonl"])
+def test_file_exists(extension):
+    tmp_file = NamedTemporaryFile(delete=False, suffix=extension)
+    tmp_path = Path(tmp_file.name)
+    try:
+        Clumper([{"a": 1}]).write_jsonl(tmp_file.name)
+    finally:
+        os.unlink(tmp_file.name)
+
+    with pytest.raises(FileExistsError):
+        Clumper([{"a": 1}]).write_jsonl(tmp_file.name)
+
+
+def test_write_csv():
+    tmp_file = NamedTemporaryFile(delete=False, suffix=".csv")
+    tmp_path = Path(tmp_file.name)
+    try:
+        Clumper([{"a": 1}, {"b": 2}]).write_csv(tmp_file.name)
+    finally:
+        os.unlink(tmp_file.name)
+
+    assert tmp_path.exists()
+    assert tmp_path.stat().st_size > 0
+
+
+def test_write_csv_overwrite():
+    tmp_file = NamedTemporaryFile(delete=False, suffix=".csv")
+    tmp_path = Path(tmp_file.name)
+    try:
+        Clumper([{"a": 1}, {"b": 2}]).write_csv(tmp_file.name)
+        Clumper([{"c": 3}]).write_csv(tmp_file.name)
+    finally:
+        os.unlink(tmp_file.name)
+
+    assert tmp_path.stat().st_size > 0
