+def package_repo(package_path, repo_path):
+    subprocess.call(["cp {}/* {} -r".format(package_path, repo_path)],
+                    shell=True)
+    subprocess.call(["git add *; git commit -am initial; git push;"],
+                    cwd=repo_path,
+                    shell=True)
