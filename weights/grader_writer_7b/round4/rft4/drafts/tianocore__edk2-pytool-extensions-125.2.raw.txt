+        """ Test fetching a git dependency """
+        env = shell_environment.GetEnvironment()
+        tmpdir = tempfile.mkdtemp(dir=test_dir)
+        try:
+            # initialize the git dependency
+            gitdep = GitDependency(TestGitDependency.TEST_DESCRIPTOR)

+            # make sure we have nothing there already
+            self.assertFalse(os.path.exists(tmpdir))
