+import pytest
+import numpy as np
+import galois
+
+
+def test_square_free_factorization():
+    # Square-free factorization of a prime power
+    p = 3
+    m = 2
+    n = 1
+    F = galois.GF(p**m)
+    factor, multiplicity = galois.square_free_factorization(F.prim_poly)
+    assert factor.degree == 1
+    assert multiplicity == [n]
+
+    # Square-free factorization of a cyclotomic coset representative
+    p = 3
+    m = 2
+    k = 2
+    F = galois.GF(p**m)
+    factor, multiplicity = galois.square_free_factorization(F.cyclotomic_cosets[k][0])
+    assert factor.degree == 1
+    assert multiplicity == [n]
+
+    # Square-free factorization of a reducible polynomial
+    p = 3
+    m = 2
+    F = galois.GF(p**m)
+    factor, multiplicity = galois.square_free_factorization((F.primitive_element**2 + F.primitive_element + 1)*(F.primitive_element + 1))
+    assert factor == [F.primitive_element**2 + F.primitive_element + 1, F.primitive_element + 1]
+    assert multiplicity == [1, 1]
