+def test_Split():
+    var = SparseRunVariable(
+        name="rt",
+        values=[1.0, 2.0],
+        run_info=RunInfo({'subject': '01'}, 15, 1, "none", 15),
+        source='events',
+        sampling_rate=1
+    )
+
+    # No grouping variables
+    coll = BIDSRunVariableCollection([var], sampling_rate=1)
+    transform.Split(coll, ["rt"], output=["a", "b"])
+    assert len(coll) == 2
+    assert coll['a'].values.tolist() == [1.0]
+    assert coll['b'].values.tolist() == [2.0]
+
+    # One grouping variable
+    coll = BIDSRunVariableCollection([var], sampling_rate=1)
+    transform.Split(coll, ["rt", "subject"], output=["a", "b"])
+    assert len(coll) == 2
+    assert coll['a'].values.tolist() == [1.0]
+    assert coll['b'].values.tolist() == [2.0]
+
+    # Two grouping variables
+    coll = BIDSRunVariableCollection([var], sampling_rate=1)
+    transform.Split(coll, ["rt", "subject", "acquisition"],
+                    output=["a", "b"])
+    assert len(coll) == 2
+    assert coll['a'].values.tolist() == [1.0]
+    assert coll['b'].values.tolist() == [2.0]
