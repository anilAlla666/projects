To address the issue where the entity ID should not be modifiable, we need to ensure that attempting to change the ID raises an error. We can achieve this by adding a property decorator to the `Entity` class that makes the `id` attribute read-only.

Here's the updated code with the necessary changes:

### rocrate/model/entity.py
```python
import uuid
from collections.abc import MutableMapping

from dateutil.parser import isoparse
from .. import vocabs


class Entity(MutableMapping):

    def __init__(self, crate, identifier=None, properties=None):
        self.crate = crate
        if identifier:
            self.id = self.format_id(identifier)
        else:
            self.id = f"#{uuid.uuid4()}"
        if properties:
            empty = self._empty()
            empty.update(properties)
            self._jsonld = empty
        else:
            self._jsonld = self._empty()

    # Format the given ID with rules appropriate for this type.
    # For example, Dataset (directory) data entities SHOULD end with /
    def format_id(self, identifier):
        return str(identifier)

    def __repr__(self):
        return f"<{self.id} {self.type}>"

    def properties(self):
        return self._jsonld

    def as_jsonld(self):
        return self._jsonld

    @property
    def _default_type(self):
        clsName = self.__class__.__name__
        if clsName in vocabs.RO_CRATE["@context"]:
            return clsName
        return "Thing"

    def canonical_id(self):
        return self.crate.resolve_id(self.id)

    def __hash__(self):
        return hash(self.canonical_id())

    def _empty(self):
        val = {
            "@id": self.id,
            "@type": self._default_type
        }
        return val

    def __getitem__(self, key):
        v = self._jsonld[key]
        if v is None or key.startswith("@"):
            return v
        values = v if isinstance(v, list) else [v]
        deref_values = []
        for entry in values:
            if isinstance(entry, dict):
                try:
                    deref_values.extend([self.crate.dereference(e) for e in entry.values()])
                except KeyError:
                    deref_values.append(entry)
            elif isinstance(entry, list):
                deref_values.extend([self.crate.dereference(e) for e in entry])
            else:
                deref_values.append(entry)
        return deref_values[0] if len(deref_values) == 1 else deref_values

    def __setitem__(self, key, value):
        raise AttributeError("Cannot modify the ID of an existing entity")

    def __delitem__(self, key):
        raise AttributeError("Cannot delete the ID of an existing entity")

    def __iter__(self):
        return iter(self._jsonld)

    def __len__(self):
        return len(self._jsonld)
```
